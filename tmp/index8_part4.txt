<script id="ux-taxonomy-manager-v2">
(function(){
  const KEY='ae-ux-intel-taxonomy-governance-v2';
  const norm=value=>String(value||'').normalize('NFD').replace(/[\u0300-\u036f]/g,'').replace(/\s+/g,' ').trim().toLocaleLowerCase('es');
  const clean=value=>String(value||'').replace(/\s+/g,' ').trim();
  const esc=value=>String(value??'').replace(/[&<>"']/g,ch=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[ch]));
  const LEVELS={product:'Producto',funnel:'Funnel',minifunnel:'Minifunnel',component:'Componente'};

  const manager=document.getElementById('taxonomy-manager-v2-modal');
  const managerTitle=document.getElementById('taxonomy-manager-v2-title');
  const managerCopy=document.getElementById('taxonomy-manager-v2-copy');
  const managerToolbar=document.getElementById('taxonomy-manager-v2-toolbar');
  const managerLevelField=document.getElementById('taxonomy-manager-v2-level-field');
  const managerLevel=document.getElementById('taxonomy-manager-v2-level');
  const managerSearch=document.getElementById('taxonomy-manager-v2-search');
  const managerList=document.getElementById('taxonomy-manager-v2-list');

  const editModal=document.getElementById('taxonomy-manager-v2-edit');

  const editInput=document.getElementById('taxonomy-manager-v2-edit-input');

  const mergeModal=document.getElementById('taxonomy-manager-v2-merge');
  const mergeMeta=document.getElementById('taxonomy-manager-v2-merge-meta');
  const mergeTarget=document.getElementById('taxonomy-manager-v2-merge-target');
  const mergeAction=document.getElementById('taxonomy-manager-v2-merge-action');

  const confirmModal=document.getElementById('taxonomy-manager-v2-confirm');
  const confirmTitle=document.getElementById('taxonomy-manager-v2-confirm-title');
  const confirmCopy=document.getElementById('taxonomy-manager-v2-confirm-copy');
  const confirmMeta=document.getElementById('taxonomy-manager-v2-meta');
  const confirmAction=document.getElementById('taxonomy-manager-v2-confirm-action');

  let activeSection='';
  let entries=[];
  let pending=null;
  let pendingEdit=null;
  let pendingMerge=null;

  function readState(){
    try{
      const state=JSON.parse(localStorage.getItem(KEY)||'null');
      if(state&&typeof state==='object')return {
        version:3,
        product:Array.isArray(state.product)?state.product:[],
        vocabularies:Array.isArray(state.vocabularies)?state.vocabularies:[],
        glossary:Array.isArray(state.glossary)?state.glossary:[],
        renames:Array.isArray(state.renames)?state.renames:[],
        merges:Array.isArray(state.merges)?state.merges:[]
      };
    }catch(_){}
    return {version:3,product:[],vocabularies:[],glossary:[],renames:[],merges:[]};
  }
  function writeState(state){
    state.version=3;
    try{localStorage.setItem(KEY,JSON.stringify(state));}catch(_){}
    try{if(typeof safeStorageWrite==='function')safeStorageWrite({project:true});}catch(_){}
  }

  function projectTerms(){
    try{
      const all=Array.isArray(taxonomyProject?.state?.terms)?taxonomyProject.state.terms:[];
      return all.filter(term=>{
        const origin=String(term?.origin||'').trim();
        return term?.governed===true||!/^excel revisado\s*·/i.test(origin);
      });
    }catch(_){return [];}
  }
  function rawTerms(){
    try{return globalThis.UXGovernedTaxonomy?.terms?.()||[];}catch(_){return [];}
  }
  function termId(term){return clean(term?.id||term?.termId||term?.key);}
  function parentId(term){return clean(term?.parentId||term?.parentTermId||term?.parent?.id||term?.parent);}
  function setParentId(term,value){
    if('parentId' in term||!('parentTermId' in term))term.parentId=value;
    else term.parentTermId=value;
  }
  function termName(term){return clean(term?.canonicalName||term?.name||term?.label||term?.value);}
  function termLevel(term){return clean(term?.level);}
  function termAliases(term){
    const raw=term?.aliases||term?.alias||term?.historicalNames||[];
    return Array.isArray(raw)?raw.map(clean).filter(Boolean):(clean(raw)?[clean(raw)]:[]);
  }
  function setAliases(term,values){
    const out=[];
    values.map(clean).filter(Boolean).forEach(value=>{if(!out.some(item=>norm(item)===norm(value)))out.push(value);});
    term.aliases=out;
  }
  function canonical(term){return ['canonical','canonico'].includes(norm(term?.status||'canonical'));}

  function applyRenameState(state,all){
    state.renames.forEach(change=>{
      const term=all.find(item=>termId(item)===change.id);
      if(!term)return;
      const current=termName(term);
      const aliases=[...termAliases(term),...(change.aliases||[]),change.originalName];
      if(current&&norm(current)!==norm(change.name))aliases.push(current);
      setAliases(term,aliases.filter(value=>norm(value)!==norm(change.name)));
      term.canonicalName=change.name;
    });
  }

  function mergeInMemory(source,target,all,visited=new Set()){
    if(!source||!target)return;
    const sourceId=termId(source),targetId=termId(target);
    if(!sourceId||!targetId||sourceId===targetId||visited.has(sourceId))return;
    visited.add(sourceId);

    setAliases(target,[...termAliases(target),termName(source),...termAliases(source)]);
    const targetChildren=()=>all.filter(item=>canonical(item)&&parentId(item)===targetId);
    const sourceChildren=all.filter(item=>canonical(item)&&parentId(item)===sourceId);

    sourceChildren.forEach(child=>{
      const duplicate=targetChildren().find(candidate=>
        termLevel(candidate)===termLevel(child)&&norm(termName(candidate))===norm(termName(child))
      );
      if(duplicate)mergeInMemory(child,duplicate,all,visited);
      else setParentId(child,targetId);
    });

    source.status='retired';
    source.mergedIntoId=targetId;
  }

  function applyMergeState(state,all){
    state.merges.forEach(change=>{
      const source=all.find(item=>termId(item)===change.sourceId);
      const target=all.find(item=>termId(item)===change.targetId);
      if(source&&target)mergeInMemory(source,target,all);
    });
  }

  function applyRetiredState(state,all){
    state.product.forEach(hidden=>{
      const term=all.find(item=>
        (hidden.id&&hidden.id===termId(item))||
        (!hidden.id&&hidden.level===termLevel(item)&&norm(hidden.name)===norm(termName(item)))
      );
      if(term&&canonical(term))term.status='retired';
    });
  }

  function applyProductGovernance(){
    const state=readState();
    const all=projectTerms();
    applyRenameState(state,all);
    applyMergeState(state,all);
    applyRetiredState(state,all);
  }

  function productTerms(){
    applyProductGovernance();
    return rawTerms().filter(term=>canonical(term)&&termLevel(term)!=='element');
  }

  function pathOf(term,terms){
    const byId=new Map(terms.map(item=>[termId(item),item]).filter(([id])=>id));
    const path=[],seen=new Set();
    let current=term;
    while(current){
      const id=termId(current);
      if(id&&seen.has(id))break;
      if(id)seen.add(id);
      if(termName(current))path.unshift(termName(current));
      current=parentId(current)?byId.get(parentId(current)):null;
    }
    return path.join(' › ');
  }

  function allRows(){
    const rows=[];
    try{
      (sourceItems||[]).forEach(item=>(item?.insightRegister?.rows||[]).forEach(row=>rows.push(row)));
      (learnings||[]).forEach(item=>rows.push(item));
    }catch(_){}
    return rows;
  }
  function matches(row,field,value){
    if(!field)return false;
    const wanted=norm(value);
    if(field==='component')return [row?.component,...(row?.components||[])].some(item=>norm(item)===wanted);
    if(field==='uxLine')return [...(row?.uxLines||[]),...String(row?.uxLine||'').split(';')].some(item=>norm(item)===wanted);
    if(field==='origin')return [...(row?.origins||[]),...String(row?.origin||'').split(/[;|]/)].some(item=>norm(item)===wanted);
    const current=row?.[field];
    if(Array.isArray(current))return current.some(item=>norm(item)===wanted);
    return norm(current)===wanted;
  }
  function usage(field,value){return allRows().filter(row=>matches(row,field,value)).length;}
  function usageForTerm(term){
    const names=[termName(term),...termAliases(term)];
    return allRows().filter(row=>names.some(name=>matches(row,termLevel(term),name))).length;
  }

  function productEntries(){
    const terms=productTerms();
    return terms.map(term=>{
      const id=termId(term),name=termName(term),level=termLevel(term);
      const children=terms.filter(item=>parentId(item)===id).length;
      return {
        section:'product',group:LEVELS[level]||'Estructura',
        id,name,level,term,path:pathOf(term,terms)||name,children,usage:usageForTerm(term),
        aliases:termAliases(term)
      };
    }).sort((a,b)=>a.path.localeCompare(b.path,'es'));
  }

  function vocabularyCards(){
    const root=document.getElementById('taxonomy-controlled');
    if(!root)return [];
    const addButtons=[...root.querySelectorAll('button')].filter(button=>['añadir','+ añadir','anadir','+ anadir'].includes(norm(button.textContent)));
    const cards=[];
    addButtons.forEach(button=>{
      let card=button.parentElement;
      while(card&&card!==root){
        const heading=[...card.querySelectorAll('h2,h3,h4,strong,[class*=title],[class*=heading]')]
          .find(node=>node!==button&&clean(node.textContent)&&norm(node.textContent)!==norm(button.textContent));
        if(heading){
          const group=clean(heading.textContent);
          if(!cards.some(item=>norm(item.group)===norm(group)))cards.push({group,card});
          break;
        }
        card=card.parentElement;
      }
    });
    return cards;
  }
  function vocabularyField(group){
    const map={
      'tipo de registro':'recordType','origen del hallazgo':'origin','dispositivo':'device','mercado':'market',
      'region':'region','estado usuario':'login','resultado':'outcome','implementacion':'implementation',
      'dimension':'uxLine','dimensión':'uxLine','dimension ux':'uxLine','impacto ux':'impactUx','sistema / plataforma':'system'
    };
    return map[norm(group)]||'';
  }
  function cardValues(group,card){
    let values=[];
    try{
      if(typeof taxonomyControlledValues==='function'){
        const source=taxonomyControlledValues(group);
        if(Array.isArray(source))values=source.map(clean).filter(Boolean);
      }
    }catch(_){}
    if(!values.length){
      values=[...card.querySelectorAll('[data-taxonomy-controlled-value],span,button')]
        .map(node=>clean(node.dataset?.taxonomyControlledValue||node.textContent))
        .filter(value=>value&&value.length<100&&norm(value)!==norm(group)&&!['añadir','+ añadir','anadir','+ anadir'].includes(norm(value)));
    }
    return [...new Map(values.map(value=>[norm(value),value])).values()];
  }
  function vocabularyEntries(){
    const hidden=readState().vocabularies;
    return vocabularyCards().flatMap(({group,card})=>
      cardValues(group,card)
        .filter(value=>!hidden.some(item=>norm(item.group)===norm(group)&&norm(item.value)===norm(value)))
        .map(value=>({section:'vocabularies',group,name:value,value,usage:usage(vocabularyField(group),value)}))
    ).sort((a,b)=>a.group.localeCompare(b.group,'es')||a.name.localeCompare(b.name,'es'));
  }

  function glossaryEntries(){
    const hidden=readState().glossary;
    const root=document.getElementById('taxonomy-glossary');
    if(!root)return [];
    const out=[];
    [...root.querySelectorAll('[data-glossary-term],article,li,details')].forEach(container=>{
      const heading=container.matches('[data-glossary-term]')?container:container.querySelector('h3,h4,strong');
      const paragraph=container.querySelector('p');
      const name=clean(container.dataset?.glossaryTerm||heading?.textContent);
      if(!name||!paragraph||hidden.some(item=>norm(item.name)===norm(name)))return;
      if(out.some(item=>norm(item.name)===norm(name)))return;
      out.push({section:'glossary',group:'Glosario',name,description:clean(paragraph.textContent),container,usage:0});
    });
    return out.sort((a,b)=>a.name.localeCompare(b.name,'es'));
  }

  function sectionEntries(section){
    if(section==='product')return productEntries();
    if(section==='vocabularies')return vocabularyEntries();
    return glossaryEntries();
  }

  function hiddenVocabulary(group,value){
    return readState().vocabularies.some(item=>norm(item.group)===norm(group)&&norm(item.value)===norm(value));
  }
  const previousControlled=globalThis.taxonomyControlledValues;
  if(typeof previousControlled==='function'){
    globalThis.taxonomyControlledValues=function(label){

  }

  function applyVocabularies(){
    const hidden=readState().vocabularies;
    if(!hidden.length)return;
    vocabularyCards().forEach(({group,card})=>{
      [...card.querySelectorAll('[data-taxonomy-controlled-value],span,button')].forEach(node=>{
        const value=clean(node.dataset?.taxonomyControlledValue||node.textContent);
        if(value&&hidden.some(item=>norm(item.group)===norm(group)&&norm(item.value)===norm(value))){
          if(!['añadir','+ añadir','anadir','+ anadir'].includes(norm(node.textContent)))node.hidden=true;

  function applyGlossary(){
    const hidden=readState().glossary;
    if(!hidden.length)return;
    const root=document.getElementById('taxonomy-glossary');
    if(!root)return;
    [...root.querySelectorAll('[data-glossary-term],article,li,details')].forEach(container=>{
      const heading=container.matches('[data-glossary-term]')?container:container.querySelector('h3,h4,strong');
      const name=clean(container.dataset?.glossaryTerm||heading?.textContent);
      if(name&&hidden.some(item=>norm(item.name)===norm(name)))container.hidden=true;
    });
  }
  function applyState(){
    ensureButtons();
    applyProductGovernance();
    applyVocabularies();
    applyGlossary();
  }

  function copyFor(section){
    if(section==='product')return ['Gestionar estructura de producto','Puedes editar el nombre, fusionar duplicados o eliminar términos creados por error. Los IDs se mantienen y los nombres anteriores se conservan como alias.'];
    if(section==='vocabularies')return ['Gestionar vocabularios','Cada valor aparece dentro de su vocabulario real. Si ya está utilizado, se retira para nuevas clasificaciones sin borrar el histórico.'];
    return ['Gestionar Glosario UX Intel','Se muestran únicamente las definiciones reales del glosario.'];
  }

  function renderList(){
    const query=norm(managerSearch.value);
    const level=activeSection==='product'?managerLevel.value:'';
    const filtered=entries.filter(entry=>{
      if(level&&entry.level!==level)return false;
      return !query||norm([entry.group,entry.name,entry.path,entry.description].filter(Boolean).join(' ')).includes(query);
    });
    if(!filtered.length){
      managerList.innerHTML='<div class="taxonomy-manager-v2-empty">No hay términos para esta búsqueda.</div>';
      return;
    }
    const groups=new Map();
    filtered.forEach(entry=>{
      const group=entry.group||'Términos';
      if(!groups.has(group))groups.set(group,[]);
      groups.get(group).push(entry);
    });
    managerList.innerHTML=[...groups.entries()].map(([group,items])=>`
      <section class="taxonomy-manager-v2-group">
        <h3>${esc(group)}</h3>
        ${items.map(entry=>{
          const index=entries.indexOf(entry);
          let meta='';
          if(entry.section==='product'){
            meta=`${entry.path}${entry.usage?` · ${entry.usage} ${entry.usage===1?'uso':'usos'}`:''}${entry.children?` · ${entry.children} ${entry.children===1?'dependencia':'dependencias'}`:''}`;
          }else if(entry.section==='vocabularies'){
            meta=entry.usage?`${entry.usage} ${entry.usage===1?'uso':'usos'} en conocimiento estructurado`:'Sin usos detectados';
          }else meta=entry.description||'Definición del glosario';

          const actions=entry.section==='product'
            ? `<div class="taxonomy-manager-v2-actions">
                <button class="is-neutral" type="button" data-taxonomy-manager-v2-edit="${index}">Editar</button>
                <button class="is-neutral" type="button" data-taxonomy-manager-v2-merge="${index}">Fusionar</button>
                <button class="is-danger" type="button" data-taxonomy-manager-v2-delete="${index}">${entry.usage?'Retirar':'Eliminar'}</button>
              </div>`
            : `<div class="taxonomy-manager-v2-actions">
                <button class="is-danger" type="button" data-taxonomy-manager-v2-delete="${index}">${entry.usage?'Retirar':'Eliminar'}</button>
              </div>`;

          return `<div class="taxonomy-manager-v2-row">
            <div class="taxonomy-manager-v2-copy"><strong>${esc(entry.name)}</strong><span>${esc(meta)}</span></div>
            ${actions}
          </div>`;
        }).join('')}
      </section>
    `).join('');
  }

  function openManager(section){
    activeSection=section;
    entries=sectionEntries(section);
    const [title,copy]=copyFor(section);
    managerTitle.textContent=title;
    managerCopy.textContent=copy;
    managerSearch.value='';
    managerLevelField.hidden=section!=='product';
    managerToolbar.classList.toggle('is-single',section!=='product');
    if(section==='product')managerLevel.value='product';
    renderList();
    manager.hidden=false;
    setTimeout(()=>managerSearch.focus(),0);
  }
  function closeManager(){manager.hidden=true;}

  function productDuplicate(entry,newName){
    return productEntries().find(candidate=>
      candidate.id!==entry.id&&candidate.level===entry.level&&norm(candidate.name)===norm(newName)
    );
  }

  function openEdit(entry){
    pendingEdit=entry;
    editMeta.textContent=entry.path;
    editInput.value=entry.name;
    editModal.hidden=false;
    setTimeout(()=>{editInput.focus();editInput.select();},0);
  }
  function closeEdit(){editModal.hidden=true;pendingEdit=null;}
  function saveEdit(){
    const entry=pendingEdit;
    if(!entry)return;
    const newName=clean(editInput.value);
    if(!newName){
      try{showToast('Nombre requerido','Escribe un nombre para el término.');}catch(_){}
      return;
    }
    if(norm(newName)===norm(entry.name)){closeEdit();return;}
    const duplicate=productDuplicate(entry,newName);
    if(duplicate){
      closeEdit();
      openMerge(entry,duplicate.id);
      try{showToast('Ese término ya existe','En lugar de duplicarlo, puedes fusionar ambos términos.');}catch(_){}
      return;
    }

    const state=readState();
    const existing=state.renames.find(item=>item.id===entry.id);
    if(existing){
      if(!existing.aliases.some(value=>norm(value)===norm(entry.name)))existing.aliases.push(entry.name);
      existing.name=newName;
      existing.updatedAt=new Date().toISOString();
    }else{
      state.renames.push({
        id:entry.id,
        originalName:entry.name,
        name:newName,
        aliases:[entry.name],
        updatedAt:new Date().toISOString()
      });
    }
    writeState(state);
    closeEdit();
    refreshAfterChange(`“${entry.name}” ahora se muestra como “${newName}”. El nombre anterior queda como alias.`);
  }

  function mergeCandidates(entry){
    return productEntries().filter(candidate=>candidate.id!==entry.id&&candidate.level===entry.level);
  }
  function openMerge(entry,preselectedId=''){
    pendingMerge=entry;
    const candidates=mergeCandidates(entry);
    mergeMeta.textContent=`Origen: ${entry.path}`;
    mergeTarget.innerHTML=candidates.length
      ? candidates.map(candidate=>`<option value="${esc(candidate.id)}">${esc(candidate.path)}</option>`).join('')
      : '<option value="">No hay otro término compatible</option>';
    if(preselectedId&&candidates.some(candidate=>candidate.id===preselectedId))mergeTarget.value=preselectedId;
    mergeAction.disabled=!candidates.length;
    mergeModal.hidden=false;
  }
  function closeMerge(){mergeModal.hidden=true;pendingMerge=null;}
  function saveMerge(){
    const source=pendingMerge;
if(!source)return;
    const target=productEntries().find(candidate=>candidate.id===mergeTarget.value);
    if(!target)return;

    const state=readState();
    state.merges=state.merges.filter(item=>item.sourceId!==source.id);
    state.merges.push({
      sourceId:source.id,
      targetId:target.id,
      sourceName:source.name,
      targetName:target.name,
      level:source.level,
      updatedAt:new Date().toISOString()
    });
    writeState(state);
    closeMerge();
    refreshAfterChange(`“${source.name}” se ha fusionado con “${target.name}”. Sus nombres históricos se conservan como alias.`);
  }

  function openConfirm(entry){
    pending=entry;
    if(entry.section==='product'&&entry.children){
      confirmTitle.textContent='Este término tiene dependencias';
      confirmCopy.textContent=`“${entry.name}” contiene ${entry.children} ${entry.children===1?'término dependiente':'términos dependientes'}. Para eliminarlo sin romper la estructura, fusiónalo con otro término del mismo nivel.`;
      confirmMeta.textContent=entry.path;
      confirmAction.textContent='Fusionar con…';
      confirmAction.className='action-button';
      pending={...entry,redirectToMerge:true};
      confirmModal.hidden=false;
      return;
    }

    confirmTitle.textContent=entry.usage?'Retirar término':'Eliminar término';
    if(entry.section==='product'){
      confirmCopy.textContent=entry.usage
        ? `“${entry.name}” ya está utilizado. Se retirará de nuevas clasificaciones, pero se conservarán las referencias históricas.`
        : `¿Quieres eliminar “${entry.name}” de la estructura activa?`;
      confirmMeta.textContent=entry.path;
    }else if(entry.section==='vocabularies'){
      confirmCopy.textContent=entry.usage
        ? `“${entry.name}” ya está utilizado. Dejará de ofrecerse en nuevas clasificaciones, sin modificar los registros existentes.`
        : `¿Quieres eliminar “${entry.name}” de “${entry.group}”?`;
      confirmMeta.textContent=entry.group;
    }else{
      confirmCopy.textContent=`¿Quieres eliminar la definición “${entry.name}” del Glosario UX Intel?`;

  function normalizeAndPruneTaxonomy(){
    try{
      const terms=Array.isArray(taxonomyProject?.state?.terms)?taxonomyProject.state.terms:null;
      if(!terms)return false;
      let changed=false;
      const byId=new Map(terms.map(term=>[termId(term),term]).filter(([id])=>id));
      const chain=term=>{
        const out=[],seen=new Set();let current=term;
        while(current){const id=termId(current);if(id&&seen.has(id))break;if(id)seen.add(id);out.unshift(current);const pid=parentId(current);current=pid?byId.get(pid):null;}
        return out;
      };

      // Retire historical Screen/Element levels and reconnect governed descendants.
      const legacyLevels=new Set(['screen','element']);
      terms.filter(term=>legacyLevels.has(termLevel(term))).forEach(legacy=>{
        const lid=termId(legacy),pid=parentId(legacy);
        terms.forEach(child=>{if(parentId(child)===lid){setParentId(child,pid);changed=true;}});
      });
      let active=terms.filter(term=>!legacyLevels.has(termLevel(term)));
      if(active.length!==terms.length)changed=true;

      // Normalize the five Booking funnels before filtering the governed taxonomy.
      const activeById=new Map(active.map(term=>[termId(term),term]).filter(([id])=>id));
      const activeChain=term=>{
        const out=[],seen=new Set();let current=term;
        while(current){const id=termId(current);if(id&&seen.has(id))break;if(id)seen.add(id);out.unshift(current);const pid=parentId(current);current=pid?activeById.get(pid):null;}
        return out;
      };
      active.forEach(term=>{
        if(termLevel(term)!=='funnel')return;
        const path=activeChain(term),product=path.find(node=>termLevel(node)==='product');
        if(norm(termName(product))!=='booking')return;
        const canonical=funnelNames.get(norm(termName(term)));if(!canonical)return;
        const oldName=termName(term);
        if(oldName&&norm(oldName)!==norm(canonical)){
          const aliases=Array.isArray(term.aliases)?[...term.aliases]:[];
          if(!aliases.some(alias=>norm(alias)===norm(oldName)))aliases.push(oldName);
          term.aliases=aliases;changed=true;
        }

      });

        if(norm(termName(funnel))!=='extras')return;

        if(oldName&&norm(oldName)!==norm(canonical)){

<script id="ux-strict-governed-taxonomy-persistence">
(function(){
  const clean=value=>String(value??'').trim();
  const norm=value=>clean(value).normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLocaleLowerCase('es');
  const legacyKey=level=>'legacy'+level.charAt(0).toUpperCase()+level.slice(1);

  function normalizeItem(item){
    if(!item||typeof item!=='object'||!globalThis.UXGovernedTaxonomy?.normalizeRecord)return false;
    const normalized=globalThis.UXGovernedTaxonomy.normalizeRecord(item);
    let changed=false;

    for(const level of ['product','funnel','minifunnel','component']){
      const before=clean(item[level]);
      const after=clean(normalized[level]);
      if(before&&norm(before)!==norm(after)&&!clean(item[legacyKey(level)])){
        item[legacyKey(level)]=before;
        changed=true;
      }
      // Un componente existente no se borra si el canon todavía no lo resuelve.
      // Se conserva como dato y queda trazado en legacyComponent para revisión.
      if(level==='component'&&before&&!after)continue;
      if(before!==after){
        item[level]=after;
        changed=true;
      }
    }

    if(Array.isArray(item.components)){
      let next=[];
      try{next=globalThis.UXGovernedTaxonomy?.componentNames?.(item)||[];}catch(_){}
      if(!next.length&&clean(normalized.component))next=[clean(normalized.component)];
      // Si no hay resolución gobernada todavía, no vaciamos un array de
      // componentes que ya contiene información válida.
      if(next.length||!item.components.length){
        if(JSON.stringify(item.components)!==JSON.stringify(next)){
          if(item.components.length&&!Array.isArray(item.legacyComponents))item.legacyComponents=[...item.components];
          item.components=next;
          changed=true;
        }
        if(next.length&&clean(item.component)!==clean(next[0])){
          if(clean(item.component)&&!clean(item.legacyComponent))item.legacyComponent=item.component;
          item.component=next[0];
          changed=true;
        }
      }
    }
    return changed;
  }

  function apply(){
    if(!globalThis.UXGovernedTaxonomy?.normalizeRecord)return false;
    let changed=false;
    const collections=[];

    collections.forEach(items=>items.forEach(item=>{
      if(normalizeItem(item))changed=true;
      const rows=item?.insightRegister?.rows;
      if(Array.isArray(rows))rows.forEach(row=>{if(normalizeItem(row))changed=true;});
    }));

    if(changed){
      try{safeStorageWrite?.({project:false});}catch(_){try{safeStorageWrite?.();}catch(__){}}
      try{globalThis.__uxSimpleExploreRefresh?.();}catch(_){}
      try{refreshAllTaxonomyFilters?.();}catch(_){}
    }
    return changed;
  }

  function init(){

    setTimeout(apply,1500);
    setTimeout(apply,3200);
  }
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',init,{once:true});
  else init();
})();

(function(){
  function clean(value){return String(value??'').trim();}

  function taxonomyTerms(){
    try{
      return Array.isArray(taxonomyProject?.state?.terms) ? taxonomyProject.state.terms : [];
    }catch(_){
      return [];
    }
  }

  function termId(term){
    return clean(term?.id || term?.termId || term?.key);
  }

  function levelOf(term){
    const level=clean(term?.level).toLowerCase();
    return ['product','funnel','minifunnel','component'].includes(level) ? level : '';
  }

  function applyLevelColors(){
    const terms=taxonomyTerms();
    if(!terms.length)return;

    const byId=new Map(
      terms.map(term=>[termId(term),term]).filter(([id])=>id)
    );

    document.querySelectorAll('#taxonomy [data-taxonomy-term]').forEach(node=>{
      const term=byId.get(clean(node.dataset.taxonomyTerm));
      const level=levelOf(term);
      if(level)node.dataset.taxonomyLevel=level;
      else delete node.dataset.taxonomyLevel;
    });
  }

  let queued=false;

    requestAnimationFrame(()=>{
      queued=false;
      applyLevelColors();
    });
  }

    schedule();

<script id="ux-taxonomy-left-tree-labels-runtime-v2">
(function(){
  function clean(value){return String(value??'').trim();}
  function terms(){
    try{return Array.isArray(taxonomyProject?.state?.terms)?taxonomyProject.state.terms:[]}catch(_){return []}
  }
  function apply(){
    const list=terms();
    if(!list.length)return;
    const map=new Map(list.map(term=>[clean(term?.id||term?.termId||term?.key),term]).filter(([id])=>id));
    document.querySelectorAll('#taxonomy #taxonomy-tree [data-taxonomy-term]').forEach(node=>{
      const term=map.get(clean(node.dataset.taxonomyTerm));
      const level=clean(term?.level).toLowerCase();
      if(['product','funnel','minifunnel','component'].includes(level)) node.dataset.taxonomyLevel=level;
    });
  }
  let queued=false;
  function schedule(){
    if(queued)return;queued=true;
    requestAnimationFrame(()=>{queued=false;apply()});
  }
  document.addEventListener('DOMContentLoaded',()=>{
    schedule();setTimeout(schedule,250);setTimeout(schedule,900);
    const tree=document.getElementById('taxonomy-tree');
    if(tree)new MutationObserver(schedule).observe(tree,{childList:true,subtree:true});
  });
  document.addEventListener('click',event=>{
    if(event.target.closest('#taxonomy [data-taxonomy-term],#taxonomy [data-taxonomy-view],[data-taxonomy-manager-v2]')){
      setTimeout(schedule,0);setTimeout(schedule,120);
    }
  });
})();
</script>
