    async function persistSourceFile(id,kind,file){
      const query=new URLSearchParams({id:String(id||''),kind:String(kind||''),name:String(file?.name||'source-file')});
      const response=await fetch(`/api/source-file?${query.toString()}`,{method:'POST',headers:{'X-UX-Intel':'source-file'},body:file});
      const payload=await response.json().catch(()=>({}));
      if(!response.ok)throw new Error(payload.error||'No se ha podido guardar el archivo fuente en el proyecto.');
      return payload;
    }
    async function getPersistentSourceBlob(id,kind){
      const query=new URLSearchParams({id:String(id||''),kind:String(kind||'')});

      return response.blob();
    }

    async function hydrateDocumentAnalyses(){
      for(const item of sourceItems){
        const beforeStatus=item.status;
        try{
          const stored=await getDocumentAnalysis(item.id);
          const fromProject=item.analysisHydration?.storage==='Proyecto/history';
          // Keep a proven project successor active without deleting or rewriting the
          // older browser artifact. Recheck here in case another tab edited it.
          if(fromProject&&stored){
            if(!sameAnalysisValue(stored,item.analysis)&&!projectReviewSubsumesLocal(item.id,item,stored).subsumed)throw Error('La revisión local ha cambiado durante la restauración. No se ha sobrescrito ninguna revisión.');
          }
          else if(stored)item.analysis=stored;
          else if(hasDocumentAnalysis(item))await saveDocumentAnalysis(item.id,item.analysis);
          else item.analysis=null;
          item.analysisHydration={storage:fromProject?'Proyecto/history':'IndexedDB/analyses',timestamp:new Date().toISOString(),found:Boolean(item.analysis),beforeStatus};
          if(!item.analysis&&item.hasAnalysis&&!documentHistoricalUnavailable(item)){
            item.analysisFailure={phase:'analysis_restore',message:'El documento conserva una referencia a su análisis, pero el análisis no está disponible en este almacenamiento.',timestamp:new Date().toISOString()};
          }
          item.status=resolvedDocumentStatus(item,true);
          item.hasAnalysis=Boolean(item.analysis)||Boolean(item.hasAnalysis);
        }catch(error){
          item.analysisHydration={storage:'IndexedDB/analyses',timestamp:new Date().toISOString(),found:false,beforeStatus,error:String(error?.message||error)};
          item.status='Error de análisis';
          console.error(`No se pudo recuperar el análisis de ${item.name}`,error);
        }
      }
      documentHydrationComplete=true;
      // Hydration never saves the analysis back or rewrites the Knowledge Base.
      // Only the document index is reconciled, preserving error/missing-analysis markers.

        const state=safeStorageRead();

          const {analysis,analysisHydration,archivedReview,...metadata}=item;

        });

    function sentenceChunks(text){

    function collectDocumentBlocks(pages){
      const blocks=[],discarded=[],sections=new Set();
      pages.forEach(page=>{
        let section=`Página ${page.page}`;
        const rawLines=String(page.text||'').split(/\n/).map(line=>line.replace(/\s+/g,' ').trim());
        const candidates=[];let buffer='';
        const flush=()=>{if(buffer.trim())candidates.push(buffer.trim());buffer='';};
        rawLines.forEach(line=>{
          if(!line){flush();return;}
          if(isHeadingLine(line)){flush();section=line.slice(0,120);sections.add(section);return;}
          if(/^[-–—•▪◦]\s*/.test(line))flush();
          buffer=`${buffer} ${line.replace(/^[-–—•▪◦]\s*/,'' )}`.trim();
          if(/[.!?]$/.test(line))flush();
        });
        flush();
        if(candidates.length<=1)candidates.splice(0,candidates.length,...sentenceChunks(page.text));
        candidates.forEach((line,index)=>{
          sentenceChunks(line.replace(/^[-–—•▪◦]\s*/,'' )).forEach((text,subIndex)=>{
            const normalized=normalizeText(text);
            if(text.length<28){discarded.push({page:page.page,text,reason:'Fragmento demasiado corto o etiqueta sin afirmación.'});return;}
            if(methodTerms.some(term=>normalized.includes(term))&&!relevanceTerms.some(term=>normalized.includes(term))){discarded.push({page:page.page,text,reason:'Contexto o metodología sin afirmación de producto.'});return;}
            const relevant=relevanceTerms.some(term=>normalized.includes(term))||/\b(debe|deberia|necesita|dificulta|facilita|problema|recomend|evitar|mantener|priorizar|funciona|no funciona)\b/.test(normalized);
            if(!relevant){discarded.push({page:page.page,text,reason:'No se detecta una afirmación aplicable a producto, UX, negocio o restricciones.'});return;}
            blocks.push({page:page.page,section,text,block:`P${page.page}-${index+1}.${subIndex+1}`});
          });
        });
      });
      const seen=new Set();
      const unique=blocks.filter(block=>{const key=normalizeText(block.text).replace(/\W/g,'');if(seen.has(key)){discarded.push({...block,reason:'Duplicado literal dentro del documento.'});return false;}seen.add(key);return true;});
      return {blocks:unique,discarded,sections:[...sections]};
    }

      return new Set(normalizeText(text).replace(/[^a-z0-9áéíóúñ]+/g,' ').split(/\s+/).filter(token=>token.length>2&&!stop.has(token)).map(token=>synonymMap[token]||token));

    function explicitContradictionSignal(text){
      const normalized=normalizeText(text);
      // Limitaciones metodológicas o reducciones positivas no son contradicciones.
      if(/\b(no demuestra|no implica|no permite atribuir|no permite concluir|pendiente de validar|no esta validada|no valida)\b/.test(normalized))return false;
      if(/\b(reduce|reducir|reduccion)\s+(?:el\s+|la\s+)?(?:esfuerzo|friccion|carga|incertidumbre|errores?)\b/.test(normalized))return false;
      const explicit=/\b(contradice|evidencia contradictoria|no confirma|no respalda|resultado negativo|direccion opuesta)\b/.test(normalized);
      const negativeOutcome=/\b(empeora|cae|disminuye|reduce)\b[^.;]{0,70}\b(conversion|avance|uso|clic|click|ctr|exito|completitud|rendimiento|reservas?)\b/.test(normalized)
        ||/\b(conversion|avance|uso|clic|click|ctr|exito|completitud|rendimiento|reservas?)\b[^.;]{0,70}\b(empeora|cae|disminuye|reduce)\b/.test(normalized);
      return explicit||negativeOutcome;
    }

    function serializableAnalysisClone(analysis){
      const issues=[],active=new WeakMap(),firstPath=new WeakMap();
      const visit=(value,path)=>{
        const type=typeof value;
        if(type==='undefined')return undefined;
        if(type==='function'||type==='symbol'||type==='bigint'){issues.push(`${path}: ${type}`);return undefined;}
        if(!value||type!=='object')return value;
        if(value instanceof Blob||value instanceof File){issues.push(`${path}: ${value.constructor.name}`);return undefined;}
        if(active.has(value)){issues.push(`${path}: referencia circular con ${active.get(value)}`);return undefined;}
        if(firstPath.has(value)){issues.push(`${path}: referencia compartida con ${firstPath.get(value)}`);return undefined;}
        active.set(value,path);firstPath.set(value,path);
        const prototype=Object.getPrototypeOf(value),plainObject=Object.prototype.toString.call(value)==='[object Object]'&&(prototype===null||value.constructor?.name==='Object');
        if(!Array.isArray(value)&&!plainObject){issues.push(`${path}: ${value.constructor?.name||'objeto no plano'}`);return undefined;}
        if(Array.isArray(value)){
          const clone=value.map((item,index)=>{
          if(typeof item==='undefined'){issues.push(`${path}.${index}: undefined en array`);return null;}
          return visit(item,`${path}.${index}`);
          });
          active.delete(value);return clone;
        }
        const clone=Object.fromEntries(Object.entries(value).flatMap(([key,item])=>{
          if(typeof item==='undefined')return [];
          return [[key,visit(item,`${path}.${key}`)]];
        }));
        active.delete(value);return clone;
      };
      const cleaned=visit(analysis,'analysis');
      if(issues.length)throw new Error(`El análisis contiene valores no serializables: ${issues.slice(0,8).join(' · ')}`);
      const json=JSON.stringify(cleaned);
      return {value:JSON.parse(json),jsonLength:json.length};
    }

    function ensureReadableStreamAsyncIterator(){
      const prototype=globalThis.ReadableStream?.prototype,symbol=globalThis.Symbol?.asyncIterator;
      if(!prototype||!symbol)return {supported:false,polyfilled:false,reason:'ReadableStream o Symbol.asyncIterator no está disponible'};
      if(typeof prototype[symbol]==='function')return {supported:true,polyfilled:false};
      Object.defineProperty(prototype,symbol,{configurable:true,writable:true,value:async function*(){
        const reader=this.getReader();
        try{for(;;){const {value,done}=await reader.read();if(done)return;yield value;}}
        finally{try{reader.releaseLock();}catch{}}
      }});
      return {supported:true,polyfilled:true};
    }
    async function readTextContentStream(page,options){
      const stream=page.streamTextContent(options),reader=stream.getReader(),content={items:[],styles:Object.create(null),lang:null};
      try{
        for(;;){
          const {value,done}=await reader.read();
          if(done)break;
          if(!value)continue;
          if(content.lang==null&&value.lang!=null)content.lang=value.lang;
          Object.assign(content.styles,value.styles||{});
          content.items.push(...(value.items||[]));
        }
      }finally{try{reader.releaseLock();}catch{}}
      return content;
    }

        if(scoped.length)candidates=scoped;
      }
      if(candidates.length)return candidates;

      // El fallback difuso solo se permite con nombres suficientemente específicos.
      // Evita que "Benchmark", "Desk research", etc. relacionen iniciativas distintas.
      if(targetBase.length<12)return [];
      candidates=sourceItems.filter(source=>sourceIdentityValues(source).some(value=>{
        const base=normalizedSourceBase(value);
        return base.length>=12&&(base.includes(targetBase)||targetBase.includes(base));
      }));
      if(item){
        const scoped=candidates.filter(source=>initiativesOverlap(item,source));
        if(scoped.length)candidates=scoped;
      }
      return candidates;
    }
    function catalogSources(item){
      const names=[...(item.sourceNames||[]),item.sourceName,item.source,...(item.publicationHistory||[]).map(entry=>entry.sourceName)].filter(value=>typeof value==='string');
      const directIds=[item.documentId,...(item.publicationHistory||[]).map(entry=>entry.documentId)].filter(Boolean);
      // These evidence records are the existing UX readings of the two named CRO reports.
      const croSources={'E-03':['S-07'],'E-04':['S-08'],'E-13':['S-07','S-08'],'AB-01':['S-07'],'AB-02':['S-08']};
      directIds.push(...(croSources[item.id]||[]));
      const catalogAliases={'Header · Síntesis A/B mobile':'S-07','Header · Síntesis A/B desktop':'S-08','Header · Evolución Above the Fold':'S-09'};
      names.forEach(name=>{if(catalogAliases[name])directIds.push(catalogAliases[name]);});
      const matchedIds=new Set(directIds);
      names.forEach(name=>sourceCandidates(name,item).forEach(source=>matchedIds.add(source.id)));
      return sourceItems.filter(source=>matchedIds.has(source.id));
    }
    function initiativesOf(item){
      if(item?.initiativeManual&&item.initiative&&!['Sin clasificar','No especificado','Transversal'].includes(item.initiative))return [item.initiative];
      const existing=[...(item.initiatives||[]),item.initiative].filter(value=>value&&!['Sin clasificar','No especificado','Transversal'].includes(value));

      const namedSources=[...(item.sourceNames||[]),item.sourceName,item.source,...(item.publicationHistory||[]).map(entry=>entry.sourceName)].filter(value=>typeof value==='string');
      const known=[...new Set(sourceItems.flatMap(source=>[...(source.initiatives||[]),source.initiative]).filter(value=>value&&!['Sin clasificar','No especificado'].includes(value)))];
      const fromNames=known.filter(name=>namedSources.some(source=>normalizeText(source).startsWith(normalizeText(name)+' ·')));
      const fromDocuments=catalogSources(item).flatMap(source=>[...(source.initiatives||[]),source.initiative]);
      const linkedLearnings=item.proof?learnings.filter(learning=>item.proof.includes(learning.id)||(learning.product===item.product&&learning.screen===item.screen&&componentsOf(learning).some(component=>componentsOf(item).includes(component)))):[];
      return [...new Set([...existing,...fromNames,...fromDocuments,...linkedLearnings.flatMap(learning=>learning.initiatives||[])].filter(value=>value&&!['Sin clasificar','No especificado','Transversal'].includes(value)))].sort((a,b)=>a.localeCompare(b,'es'));
    }
    function catalogCollection(group){
      return {learning:learnings,evidence:evidenceItems,experiment:experiments,pattern:patternItems,source:sourceItems,explore:[...learnings,...evidenceItems,...experiments,...patternItems,...knowledgeUnits,...openQuestionItems,...sourceItems]}[group]||[];
    }
    const initiativeExcludedValues=new Set(['sin clasificar','no especificado','transversal','documentacion']);
    const initiativeNameEquals=(value,target)=>Boolean(String(value||'').trim()&&String(target||'').trim()&&normalizeText(value)===normalizeText(target));
    function dedupeInitiativeNames(values){
      const seen=new Set(),out=[];
      (values||[]).map(value=>String(value||'').trim()).filter(Boolean).forEach(value=>{const key=normalizeText(value);if(initiativeExcludedValues.has(key)||seen.has(key))return;seen.add(key);out.push(value);});
      return out;
    }
    function replaceInitiativeList(values,oldName,newName){
      return dedupeInitiativeNames((Array.isArray(values)?values:[]).map(value=>initiativeNameEquals(value,oldName)?newName:value));
    }
    function syncInsightRegisterInitiative(item){
      if(!item?.insightRegister||!Array.isArray(item.insightRegister.rows))return false;
      const initiatives=dedupeInitiativeNames([...(item.initiatives||[]),item.initiative]);
      const primary=String(item.initiative||initiatives[0]||'').trim();
      if(!primary)return false;
      let changed=false;
      item.insightRegister.rows.forEach(row=>{
        if(!row||typeof row!=='object')return;
        if(row.initiative!==primary){row.initiative=primary;changed=true;}
        if(Array.isArray(row.initiatives)){const next=[...initiatives];if(JSON.stringify(row.initiatives)!==JSON.stringify(next)){row.initiatives=next;changed=true;}}
        if(row.extraFields&&typeof row.extraFields==='object'){
          Object.keys(row.extraFields).forEach(key=>{if(['iniciativa','initiative'].includes(normalizeText(key))&&row.extraFields[key]!==primary){row.extraFields[key]=primary;changed=true;}});
        }
      });
      return changed;
    }
    function updateInitiativeMetadata(target,oldName,newName){
      if(!target||typeof target!=='object')return false;
      let changed=false;
      if(initiativeNameEquals(target.initiative,oldName)){target.initiative=newName;changed=true;}
      if(Array.isArray(target.initiatives)){const next=replaceInitiativeList(target.initiatives,oldName,newName);if(JSON.stringify(next)!==JSON.stringify(target.initiatives)){target.initiatives=next;changed=true;}}
      if(target.documentProfile&&initiativeNameEquals(target.documentProfile.initiative,oldName)){target.documentProfile.initiative=newName;changed=true;}
      if(target.metadata&&initiativeNameEquals(target.metadata.initiative,oldName)){target.metadata.initiative=newName;changed=true;}
      if(target.analysis?.metadata&&initiativeNameEquals(target.analysis.metadata.initiative,oldName)){target.analysis.metadata.initiative=newName;changed=true;}
      if(target.insightRegister&&Array.isArray(target.insightRegister.rows)){
        target.insightRegister.rows.forEach(row=>{
          if(!row||typeof row!=='object')return;
          if(initiativeNameEquals(row.initiative,oldName)){row.initiative=newName;changed=true;}
          if(Array.isArray(row.initiatives)){const next=replaceInitiativeList(row.initiatives,oldName,newName);if(JSON.stringify(next)!==JSON.stringify(row.initiatives)){row.initiatives=next;changed=true;}}
          if(row.extraFields&&typeof row.extraFields==='object'){Object.keys(row.extraFields).forEach(key=>{if(['iniciativa','initiative'].includes(normalizeText(key))&&initiativeNameEquals(row.extraFields[key],oldName)){row.extraFields[key]=newName;changed=true;}});}
        });
      }
      return changed;
    }
