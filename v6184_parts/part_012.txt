    function uxRegisterNorm(value){
      return String(value??'')
        .normalize('NFD').replace(/[\u0300-\u036f]/g,'')
        .replace(/\s+/g,' ').trim().toLocaleLowerCase('es');
    }
    function uxRegisterSplit(value){
      return String(value||'').split(/[;|·\n]/).map(part=>uxRegisterNorm(part)).filter(Boolean);
    }
    function validateInsightRegisterControlledVocabulary(parsed){
      const invalid={recordType:new Set(),origin:new Set(),uxLine:new Set()};
      let tooManyDimensions=0;
      (parsed?.rows||[]).forEach(row=>{
        const normalizedRecordType=(typeof normalizeRecordType==='function')?normalizeRecordType(row?.recordType):row?.recordType;
        const recordType=uxRegisterNorm(normalizedRecordType);
        if(recordType&&!UX_INTEL_ALLOWED_RECORD_TYPES.has(recordType))invalid.recordType.add(String(row.recordType));
        uxRegisterSplit(row?.origin).forEach(value=>{
          if(!UX_INTEL_ALLOWED_ORIGINS.has(value))invalid.origin.add(value);
        });
        const dimensions=uxRegisterSplit(row?.uxLine);
        if(dimensions.length>2)tooManyDimensions++;
        dimensions.forEach(value=>{
          if(!UX_INTEL_ALLOWED_UX_DIMENSIONS.has(value))invalid.uxLine.add(value);
        });
      });
      const messages=[];
      if(invalid.recordType.size)messages.push(`Tipo de registro fuera de vocabulario: ${[...invalid.recordType].join(', ')}`);
      if(invalid.origin.size)messages.push(`Origen del hallazgo fuera de vocabulario: ${[...invalid.origin].join(', ')}`);
      if(invalid.uxLine.size)messages.push(`Dimensión UX fuera de vocabulario: ${[...invalid.uxLine].join(', ')}`);
      if(tooManyDimensions)messages.push(`Dimensión UX admite un máximo de 2 valores (${tooManyDimensions} ${tooManyDimensions===1?'fila':'filas'})`);
      if(messages.length)throw new Error(messages.join('. '));
      return true;
    }
    function insightRegisterGovernedStructure(rows){
      const levels=['product','funnel','minifunnel','component'];
      const result={};
      levels.forEach(level=>result[level]=new Set());
      (rows||[]).forEach(row=>{
        levels.forEach(level=>{
          const values=level==='component'
            ? [...(Array.isArray(row?.components)?row.components:[]),row?.component]
            : [row?.[level]];
          values.map(value=>String(value||'').trim()).filter(Boolean)
            .forEach(value=>result[level].add(uxRegisterNorm(value)));
        });
      });
      return result;
    }
    function insightRegisterIntroducesGovernedStructure(parsed,previous){
      if(!previous)return true;
      const before=insightRegisterGovernedStructure(previous?.rows||[]);
      const after=insightRegisterGovernedStructure(parsed?.rows||[]);
      return Object.keys(after).some(level=>[...after[level]].some(value=>!before[level].has(value)));
    }

    async function applyInsightRegisterTaxonomyForUpdate(parsed,previous,fileName){
      validateInsightRegisterControlledVocabulary(parsed);
      // En una sustitución editorial del mismo Excel, si la estructura gobernada
      // no incorpora ningún valor nuevo, no se vuelve a crear/retirar taxonomía.
      // Esto evita que un validador taxonómico legacy bloquee el guardado.
      if(previous&&!insightRegisterIntroducesGovernedStructure(parsed,previous)){
        return {skipped:true,reason:'same-governed-structure'};
      }
      const taxonomyResult=await applyInsightRegisterTaxonomySafely(parsed,fileName);
      if(taxonomyResult===null)return null;

    function searchTokens(query){
      const normalized=normalizeText(query)
        .replace(/[^\p{L}\p{N}+\-/]+/gu,' ')
        .replace(/\s+/g,' ')
        .trim();
      if(!normalized)return [];
      const raw=normalized.split(' ').filter(Boolean);
      return [...new Set(raw.filter(token=>token.length>1&&!searchStopWords.has(token)))];
    }

    function expandSearchTokens(tokens){
      const expanded=new Set(tokens);
      const normalizedGroups=searchSynonymGroups.map(group=>group.map(normalizeText));
      tokens.forEach(token=>{
        normalizedGroups.forEach(group=>{
          if(group.some(term=>term===token||term.split(' ').includes(token))){
            group.forEach(term=>term.split(' ').forEach(piece=>{if(piece.length>1&&!searchStopWords.has(piece))expanded.add(piece);}));
          }
        });
      });
      return [...expanded];
    }

    function searchableArray(value){
      if(Array.isArray(value))return value.flatMap(searchableArray);
      if(value&&typeof value==='object')return Object.values(value).flatMap(searchableArray);
      return [String(value??'')];
    }

    function searchBlob(parts){
      return normalizeText(searchableArray(parts).join(' ')).replace(/\s+/g,' ').trim();
    }

    function tokenCoverage(blob,tokens){
      if(!tokens.length)return 0;
      const matched=tokens.filter(token=>blob.includes(token)).length;
      return matched/tokens.length;
    }

    function weightedScore(query,tokens,sections){
      const nq=normalizeText(query).trim();
      let score=0;
      let covered=new Set();
      sections.forEach(({value,weight=1})=>{
        const blob=searchBlob(value);
        if(!blob)return;
        if(nq&&blob.includes(nq))score+=weight*4;
        tokens.forEach(token=>{
          if(blob.includes(token)){score+=weight;covered.add(token);}
        });
      });
      if(tokens.length){
        const coverage=covered.size/tokens.length;
        if(coverage===1)score+=18;
        else if(coverage>=.67)score+=7;
        else if(coverage<.5)score-=4;
      }
      return score;
    }

    function searchOptions(popover){return [...popover.querySelectorAll('[data-search-kind]')];}
    function focusSearchOption(popover,index){
      const options=searchOptions(popover);
      if(!options.length)return;
      const safeIndex=(index+options.length)%options.length;
      options.forEach((option,i)=>option.tabIndex=i===safeIndex?0:-1);
      options[safeIndex].focus();
    }
    function searchKeyboard(event,input,popover){
      if(event.key==='Escape'){closeSearch();input.focus();return;}
      const options=searchOptions(popover);
      if(!options.length)return;
      const current=options.indexOf(document.activeElement);
      if(event.key==='ArrowDown'){
        event.preventDefault();
        focusSearchOption(popover,current<0?0:current+1);
      }else if(event.key==='ArrowUp'){
        event.preventDefault();
        focusSearchOption(popover,current<0?options.length-1:current-1);
      }else if(event.key==='Enter'&&document.activeElement===input&&popover.classList.contains('open')){
        event.preventDefault();
        options[0]?.click();
      }
    }

      input.setAttribute('aria-expanded','true');
    }

  function updateBaseCounters(){
    // V6: Resumen ya calcula sus métricas con la semántica actual
    // (registros, Excel revisados, PDF de apoyo e iniciativas). No sobrescribirlas
    // con los contadores heredados de la arquitectura anterior.
    try{ if(typeof renderOverview==='function') renderOverview(); }catch(e){}
  }

  function text(v){return String(v??'').replace(/\s+/g,' ').trim();}
  function csvCell(v){return '"' + text(v).replace(/"/g,'""') + '"';}

  function exportExplore(){
    const root=document.getElementById('explore-content');
    if(!root){ alert('No hay resultados disponibles para exportar.'); return; }
    const candidates=[...root.querySelectorAll('article, details, .cluster-card, .compact-item, .knowledge-card, [data-detail], [data-learning]')];
    const rows=[];
    const seen=new Set();
    for(const el of candidates){
      const t=text(el.innerText);
      if(!t || t.length<20 || seen.has(t)) continue;
      seen.add(t);
      rows.push([t]);
    }
    if(!rows.length){
      alert('No hay resultados visibles para exportar con los filtros actuales.');
      return;
    }
    const csv=[['Resultado visible'],...rows].map(r=>r.map(csvCell).join(';')).join('\n');
    const blob=new Blob(["\ufeff"+csv],{type:'text/csv;charset=utf-8;'});
    const url=URL.createObjectURL(blob);
    const a=document.createElement('a');
    a.href=url;
    a.download='UX_Intelligence_resultados_filtrados.csv';
    a.click();
    setTimeout(()=>URL.revokeObjectURL(url),1000);
  }

  document.addEventListener('click',e=>{
    const jump=e.target.closest('[data-panel-jump]');
    if(jump && typeof showPanel==='function'){
      e.preventDefault();
      showPanel(jump.dataset.panelJump);
    }
    if(e.target.closest('#export-explore-xlsx')) exportExplore();
  });

  window.addEventListener('load',()=>setTimeout(updateBaseCounters,250));
  setTimeout(updateBaseCounters,700);
})();
