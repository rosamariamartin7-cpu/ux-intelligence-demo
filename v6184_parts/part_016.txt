  function saveEdit(){
    const entry=pendingEdit;
    if(!entry)return;
    const newName=clean(editInput.value);
    if(!newName){
      try{showToast('Nombre requerido','Escribe un nombre para el término.');}catch(_){}
      return;
    }
    if(norm(newName)===norm(entry.name)){closeEdit();return;}
    const duplicate=productDuplicate(entry,newName);
    if(duplicate){
      closeEdit();
      openMerge(entry,duplicate.id);
      try{showToast('Ese término ya existe','En lugar de duplicarlo, puedes fusionar ambos términos.');}catch(_){}
      return;

    const state=readState();
    const existing=state.renames.find(item=>item.id===entry.id);
    if(existing){
      if(!existing.aliases.some(value=>norm(value)===norm(entry.name)))existing.aliases.push(entry.name);
      existing.name=newName;
      existing.updatedAt=new Date().toISOString();
    }else{
      state.renames.push({
        id:entry.id,
        originalName:entry.name,
        name:newName,
        aliases:[entry.name],
        updatedAt:new Date().toISOString()
      });
    }
    writeState(state);
    closeEdit();
    refreshAfterChange(`“${entry.name}” ahora se muestra como “${newName}”. El nombre anterior queda como alias.`);

    if(!target)return;

    state.merges=state.merges.filter(item=>item.sourceId!==source.id);

  function perform(){
    const entry=pending;
    if(!entry)return;
    if(entry.redirectToMerge){
      closeConfirm();
      openMerge(entry);
      return;
    }

    const state=readState();
    if(entry.section==='product'){
      if(!state.product.some(item=>item.id===entry.id&&entry.id))state.product.push({id:entry.id,name:entry.name,level:entry.level});
      if(entry.term)entry.term.status='retired';

      if(!state.vocabularies.some(item=>norm(item.group)===norm(entry.group)&&norm(item.value)===norm(entry.name)))state.vocabularies.push({group:entry.group,value:entry.name});
    }else{
      if(!state.glossary.some(item=>norm(item.name)===norm(entry.name)))state.glossary.push({name:entry.name});
    }
    writeState(state);
    closeConfirm();
    refreshAfterChange(entry.usage?'El término se ha retirado de nuevas clasificaciones. El histórico se conserva.':'El término se ha eliminado de la taxonomía activa.');
  }

(function(){
  const clean=value=>String(value??'').trim();
  const norm=value=>clean(value).normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLocaleLowerCase('es');
  const legacyKey=level=>'legacy'+level.charAt(0).toUpperCase()+level.slice(1);

  function normalizeItem(item){
    if(!item||typeof item!=='object'||!globalThis.UXGovernedTaxonomy?.normalizeRecord)return false;
    const normalized=globalThis.UXGovernedTaxonomy.normalizeRecord(item);
    let changed=false;

    for(const level of ['product','funnel','minifunnel','component']){
      const before=clean(item[level]);
      const after=clean(normalized[level]);
      if(before&&norm(before)!==norm(after)&&!clean(item[legacyKey(level)])){
        item[legacyKey(level)]=before;
        changed=true;
      }
      // Un componente existente no se borra si el canon todavía no lo resuelve.
      // Se conserva como dato y queda trazado en legacyComponent para revisión.
      if(level==='component'&&before&&!after)continue;
      if(before!==after){
        item[level]=after;
        changed=true;
      }
    }

    if(Array.isArray(item.components)){
      let next=[];
      try{next=globalThis.UXGovernedTaxonomy?.componentNames?.(item)||[];}catch(_){}
      if(!next.length&&clean(normalized.component))next=[clean(normalized.component)];
      // Si no hay resolución gobernada todavía, no vaciamos un array de
      // componentes que ya contiene información válida.
      if(next.length||!item.components.length){
        if(JSON.stringify(item.components)!==JSON.stringify(next)){
          if(item.components.length&&!Array.isArray(item.legacyComponents))item.legacyComponents=[...item.components];
          item.components=next;
          changed=true;
        }
        if(next.length&&clean(item.component)!==clean(next[0])){
          if(clean(item.component)&&!clean(item.legacyComponent))item.legacyComponent=item.component;
          item.component=next[0];
          changed=true;
        }
      }
    }
    return changed;
  }

  function apply(){

    }));

      try{globalThis.__uxSimpleExploreRefresh?.();}catch(_){}
      try{refreshAllTaxonomyFilters?.();}catch(_){}
