    function reconcileInitiativesFromSources(){
      const liveSourceInitiatives=dedupeInitiativeNames(sourceItems.flatMap(item=>[...(item.initiatives||[]),item.initiative]));
      const liveKeys=new Set(liveSourceInitiatives.map(normalizeText));
      let changed=false;
      learnings.forEach(item=>{
        const linked=dedupeInitiativeNames(catalogSources(item).flatMap(source=>[...(source.initiatives||[]),source.initiative]));
        if(!linked.length)return;
        const current=dedupeInitiativeNames([...(item.initiatives||[]),item.initiative]);
        const orphaned=current.filter(value=>!liveKeys.has(normalizeText(value)));
        if(!orphaned.length)return;
        const next=dedupeInitiativeNames([...current.filter(value=>liveKeys.has(normalizeText(value))),...linked]);
        if(!next.length)return;
        if(JSON.stringify(item.initiatives||[])!==JSON.stringify(next)){item.initiatives=next;changed=true;}
        if(!next.some(value=>initiativeNameEquals(value,item.initiative))){item.initiative=next[0];changed=true;}
      });
      return changed;
    }

    function hierarchyFilter(items,group){
      const product=document.getElementById(`${group}-product`)?.value||'Todos';
      const funnel=document.getElementById(`${group}-funnel`)?.value||'Todos';
      const minifunnel=document.getElementById(`${group}-minifunnel`)?.value||'Todos';

      const initiative=document.getElementById(`${group}-initiative`)?.value||'Todos';
      const exact=(actual,selected)=>['Todos','Todas',''].includes(selected)||normalizeText(actual||'')===normalizeText(selected);

      const contextMatches=item=>{
        const current=governedTaxonomyRecord(item);
        return exact(current.product,product)&&
          exact(current.funnel,funnel)&&
          exact(current.minifunnel,minifunnel)&&
          (['Todos','Todas'].includes(component)||componentsOf(item).some(value=>exact(value,component)));
      };

      return items.filter(item=>{
        const itemInitiatives=group==='learning'?governedLearningInitiatives(item):[...(item.initiatives||[]),item.initiative].filter(Boolean);
        const initiativeMatch=['Todos','Todas'].includes(initiative)||itemInitiatives.some(value=>exact(value,initiative));
        if(!initiativeMatch)return false;

        // Un documento puede abarcar varios contextos. Basta con que uno de sus
        // registros estructurados coincida con el filtro seleccionado.
        if(group==='source'){
          const candidates=[item,...(item?.insightRegister?.rows||[])];
          return candidates.some(contextMatches);
        }
        return contextMatches(item);
      });
    }

    function homeQuestions(){
      // La Home no es un backlog de todas las preguntas históricas.
      // Muestra únicamente preguntas de learnings consolidados, una por learning,
      // priorizadas y con diversidad de iniciativas.
      const rank={Alto:0,Medio:1,Bajo:2};
      const candidates=learnings
        .filter(learning=>Array.isArray(learning.open)&&learning.open.some(value=>String(value||'').trim()))
        .map(learning=>{
          const question=(learning.open||[]).map(value=>String(value||'').trim()).find(Boolean)||'';
          const initiative=(governedLearningInitiatives(learning)[0]||learning.initiative||'Sin iniciativa').trim();
          const updated=Date.parse(learning.updatedAt||learning.createdAt||'')||0;
          return {
            text:question,
            item:learning,
            action:`data-learning="${escapeHtml(learning.id)}"`,
            sources:catalogSources(learning),
            initiative,
            impactRank:rank[String(learning.impactUx||'').trim()]??3,
            updated
          };
        })
        .filter(candidate=>candidate.text)
        .sort((a,b)=>a.impactRank-b.impactRank||b.updated-a.updated||a.text.localeCompare(b.text,'es'));

      const selected=[];
      const perInitiative=new Map();
      for(const candidate of candidates){
        if(selected.some(existing=>semanticSimilarity(existing.text,candidate.text)>=.72))continue;
        const key=normalizeText(candidate.initiative||'Sin iniciativa');
        const used=perInitiative.get(key)||0;
        if(used>=2)continue;
        selected.push(candidate);
        perInitiative.set(key,used+1);
        if(selected.length>=8)break;
      }
      return selected;
    }

    function realActivity(){
      const events=[];
      sourceItems.forEach(item=>{
        const published=item.analysis?.publication;
        if(published?.status==='published'&&published.publishedAt)events.push({date:published.publishedAt,title:item.name,description:'Conocimiento publicado',action:`data-doc-action="open" data-id="${escapeHtml(item.id)}"`});

    function inclusionTimestamp(value){
      const raw=String(value||'').trim();
      if(!raw)return NaN;
      const parsed=Date.parse(/^\d{4}-\d{2}-\d{2}$/.test(raw)?`${raw}T12:00:00`:raw);
      return Number.isFinite(parsed)?parsed:NaN;
    }
    function inclusionDateMatch(value,from='',to=''){
      if(!from&&!to)return true;
      const time=inclusionTimestamp(value);
      if(!Number.isFinite(time))return false;
      const fromTime=from?Date.parse(`${from}T00:00:00`):-Infinity;
      const toTime=to?Date.parse(`${to}T23:59:59.999`):Infinity;
      return time>=fromTime&&time<=toTime;
    }
    function sortByInclusion(items,direction='newest',getter=item=>item?.ingestedAt||item?.createdAt||''){
      const sign=direction==='oldest'?1:-1;
      return [...items].sort((a,b)=>{
        const ta=inclusionTimestamp(getter(a)),tb=inclusionTimestamp(getter(b));
        const aKnown=Number.isFinite(ta),bKnown=Number.isFinite(tb);
        if(aKnown&&bKnown&&ta!==tb)return sign*(ta-tb);
        if(aKnown!==bKnown)return aKnown?-1:1;
        return 0;
      });
    }

    function reviewStructuralFragment(text){
      const clean=String(text||'').trim(),value=normalizeText(clean);
      if(!clean)return 'Registro vacío.';
      if(/:\s*$/.test(clean))return 'Introducción o enumeración incompleta; se usa como contexto del contenido posterior.';
      if(!/[.!?]$/.test(clean)&&clean.split(/\s+/).length<=8&&!hasSemanticPredicate(clean))return 'Título o etiqueta sin una afirmación completa.';
      if(/^(vertical|cualquier elemento)\b/.test(value)&&(clean.match(/,/g)||[]).length>=2)return 'Lista usada como contexto de jerarquía, sin una proposición independiente.';
      if(!hasSemanticPredicate(clean)&&((clean.match(/,/g)||[]).length>=2||/^(pasajeros|claim\s+banners)\b/.test(value)))return 'Lista o fragmento sin una proposición semántica completa.';
      return '';
    }
    function segmentTableColumn(lines,expected){
      const groups=[];let current=[];
      const startsCell=line=>/^["“'¿A-ZÁÉÍÓÚÑ0-9]/.test(line);
      lines.forEach(line=>{
        const previous=current.at(-1)||'',continued=/\b(y|e|o|de|del|la|el|con|para|por|en|aunque|pero|sin)$/i.test(previous.trim());
        if(current.length&&startsCell(line)&&!continued){groups.push(current.join(' ').replace(/\s+/g,' ').trim());current=[];}
        current.push(line);
      });
      if(current.length)groups.push(current.join(' ').replace(/\s+/g,' ').trim());
      while(groups.length>expected){let index=1;for(let i=2;i<groups.length;i++)if(groups[i].length<groups[index].length)index=i;groups[index-1]=`${groups[index-1]} ${groups[index]}`;groups.splice(index,1);}
      if(groups.length!==expected){
        const balanced=[];let cursor=0;
        for(let index=0;index<expected;index++){const remaining=lines.length-cursor,slots=expected-index,size=Math.max(1,Math.round(remaining/slots));balanced.push(lines.slice(cursor,cursor+size).join(' ').replace(/\s+/g,' ').trim());cursor+=size;}
        return balanced;
      }

    function strictReviewType(finding,pageContext=''){
      const text=normalizeText(finding.text),section=normalizeText(finding.source?.section||''),context=`${text} ${normalizeText(pageContext)}`;
      const authorConclusion=/\b(conclusiones|insights|key findings)\b/.test(section);
      const documentedRestriction=/\b(requisito|condicionante|obligatoriamente|limita las opciones)\b/.test(text)&&/\b(seo|h1|copy|palabras)\b/.test(context);
      const futureValidation=/\b(debera|deberia|tendria que|habra que|siguiente fase|antes de implementar|validarse|testar|testarlas|medirse|contrastar|apoyarse en datos)\b/.test(text)&&/\b(test|cro|research|datos|medir|medirse|validar|validarse|hipotesis|variantes)\b/.test(text);
      const actualExperiment=/\b(test a\/?b|experimento|variante|control)\b/.test(text)&&/\b(resultado|obtuvo|mostro|aumento|disminuyo|uplift|chance to win|significancia|supero)\b/.test(text)&&(/%|\d/.test(text));
      const uxEvaluation=/\b(desde (?:una )?(?:lectura )?ux|se percibe|es correcto desde ux|aspiracional|accionable|carga cognitiva|peso visual|puede (?:reducir|aumentar|aportar|perder|desviar|no activar)|pueden necesitar|son utiles,? pero|conviene|parece)\b/.test(text);
      const opportunity=/\b(oportunidad|tendria sentido explorar|deberia priorizar|deberia trabajarse|deberia documentarse|aplicar enfoque|simplificar|reforzar su protagonismo|coordinar su evolucion|optimizar|se explora otra forma)\b/.test(text);
      const explicitObservation=/^(desktop|mobile):/.test(text)&&/\b(ocupa|aparece|se parte|anade|suma|esta|tiene)\b/.test(text);
      if(documentedRestriction)return 'Hecho';
      if(actualExperiment)return 'Experimento';
      if(futureValidation||/\binteresa saber si\b/.test(text))return 'Hipótesis';
      if(authorConclusion)return 'Insight del autor';
      if(opportunity)return 'Oportunidad';
      if(uxEvaluation)return 'Inferencia';
      if(explicitObservation)return 'Hecho';
      if(finding.type==='Experimento'||finding.type==='Decisión')return 'Hallazgo';
      if(finding.type==='Hecho'&&!/\b(deberia|puede|conviene|se percibe)\b/.test(text))return 'Hecho';

    function formatPageRanges(values){
      const pages=[...new Set(values.filter(Number.isFinite))].sort((a,b)=>a-b),ranges=[];let start=null,previous=null;
      pages.forEach(page=>{if(start===null){start=previous=page;return;}if(page===previous+1){previous=page;return;}ranges.push(start===previous?`${start}`:`${start}–${previous}`);start=previous=page;});
      if(start!==null)ranges.push(start===previous?`${start}`:`${start}–${previous}`);
      return ranges.join(', ')||'No especificadas';
    }
    function reviewExplicitTopicId(finding){
      const text=normalizeText(finding.text),structure=normalizeText(finding.structure?.area||''),tags=normalizeText((finding.contextTags||[]).join(' '));
      const tableArea={buscador:'search-priority',cta:'cta-form',claim:'claim-seo',header:'header-account',banners:'banners','accesos rapidos':'self-service',mobile:'mobile','ui kit':'design-system'}[structure];
      if(tableArea)return tableArea;
      let anchored='';
      if(/\b(ultima busqueda|busqueda guardada|retomar el proceso)\b/.test(text))anchored='last-search';
      else if(/\b(primer bloque de la home concentra|convive con elementos|ordena mejor la convivencia|muchas funciones necesarias|demasiadas capas)\b/.test(text))anchored='above-fold';
      else if(/\b(check-?\s*in|gestiona (?:tu )?reserva|informacion de vuelos|autoservicio|accesos (?:rapidos|directos)|cards de autoservicio|tareas frecuentes|acciones (?:muy )?pulsadas|resolver una tarea concreta|responden a acciones que los usuarios)\b/.test(text))anchored='self-service';
      else if(/\b(claim|h1|seo)\b/.test(text)||tags.includes('restriccion seo'))anchored='claim-seo';
      else if(/\b(banner|campana|promocion|carrusel|magnolia|ctr|revenue)\b/.test(text))anchored='banners';
      else if(/\b(ui kit|componente robusto|deuda visual|variantes,? estados|accesibilidad bien definida)\b/.test(text))anchored='design-system';
      else if(/\b(mobile-first|escalar(?:lo)? a desktop|escala(?:r)? (?:el buscador )?a desktop)\b/.test(text))anchored='mobile';
      else if(/\b(siguiente fase|contrastar estos hallazgos|apoyandonos en datos|formular hipotesis|testarlas con cro)\b/.test(text))anchored='measurement';
      else if(/\b(cta|buscar vuelos|campos clave|formulario|que debe completar)\b/.test(text))anchored='cta-form';
      else if(/\b(header|cuenta|suma|navegacion global|menu)\b/.test(text)&&!/\babove the fold\b/.test(text))anchored='header-account';
      else if(/\b(above the fold|primer impacto|demasiadas capas|jerarquia (?:comun|visual)|accion principal)\b/.test(text))anchored='above-fold';
      return anchored;
    }
