    while(current && current!==stopAt){
      const parent=current.parentElement;
      const hasMeaningfulText=String(current.textContent||'').trim().length>0;
      const hasMeaningfulChildren=[...current.children].some(child=>{
        if(child.matches('script,style'))return false;
        return String(child.textContent||'').trim().length>0 ||
               child.matches('button,input,select,textarea,table,ul,ol,details');
      });

      if(!hasMeaningfulText && !hasMeaningfulChildren){
        current.remove();
      }
      current=parent;
    }
  }

  function mergeGlossaryIntro(){
    const panel=document.getElementById('taxonomy-view-glossary');
    const root=document.getElementById('taxonomy-glossary');
    if(!panel||!root)return;

    const intro=panel.querySelector(':scope > .ux-taxonomy-view-intro');
    if(!intro)return;

    const isCreateTermButton=button=>{
      const text=norm(button?.textContent);
      return text.includes('crear término') || text.includes('crear termino');
    };

    // El glosario se vuelve a renderizar al editar un término.
    // taxonomy.js crea entonces un botón nuevo dentro de #taxonomy-glossary.
    // Antes se movía cada botón nuevo a la cabecera sin retirar el anterior,
    // por eso "+ Crear término" se multiplicaba.
    const freshButtons=[...root.querySelectorAll('button')].filter(isCreateTermButton);
    const freshButton=freshButtons[freshButtons.length-1]||null;

    let actions=intro.querySelector(':scope > .ux-taxonomy-header-actions');
    if(!actions){
      actions=document.createElement('div');
      actions.className='ux-taxonomy-header-actions';
      [...intro.children]
        .filter(child=>child.tagName==='BUTTON')
        .forEach(button=>actions.appendChild(button));
      intro.appendChild(actions);
    }

    if(freshButton){
      // Conservamos siempre el botón recién creado por taxonomy.js porque
      // contiene los listeners actuales. Cualquier versión anterior se elimina.
      [...intro.querySelectorAll('button')]
        .filter(button=>isCreateTermButton(button))
        .forEach(button=>button.remove());

      const oldParent=freshButton.parentElement;
      actions.appendChild(freshButton);
      pruneEmptyAncestors(oldParent,root);

      // Si por un re-render intermedio aparecieron varios, solo sobrevive el último.
      freshButtons.slice(0,-1).forEach(button=>{
        const parent=button.parentElement;
        button.remove();
        pruneEmptyAncestors(parent,root);
      });
    }else{
      // Estado estable: nunca puede haber más de un "+ Crear término" en cabecera.
      const headerButtons=[...intro.querySelectorAll('button')].filter(isCreateTermButton);
      headerButtons.slice(1).forEach(button=>button.remove());
      if(headerButtons[0]&&headerButtons[0].parentElement!==actions){
        actions.appendChild(headerButtons[0]);
      }
    }

    const candidates=[...root.querySelectorAll('section,article,div,header')];
    const duplicate=candidates
      .filter(node=>{
        const text=norm(node.textContent);
        return text.includes('glosario ux intel') &&
               text.includes('fuente de verdad editable para el lenguaje de la plataforma.');
      })
      .sort((a,b)=>a.querySelectorAll('*').length-b.querySelectorAll('*').length)[0];

    if(duplicate)duplicate.remove();

    [...root.children].forEach(child=>{
      const text=String(child.textContent||'').trim();
      const interactive=child.querySelector('button,input,select,textarea,table,ul,ol,details');
      if(!text && !interactive)child.remove();
    });
  }

  function patch(){
    if(patching)return;
    patching=true;
    try{
      removeVocabularyDuplicate();
      mergeGlossaryIntro();
    }finally{
      patching=false;
    }
  }

  function observe(){
    patch();
    const taxonomy=document.getElementById('taxonomy');
    if(!taxonomy)return;
    new MutationObserver(()=>patch()).observe(taxonomy,{childList:true,subtree:true});
  }

  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded',()=>setTimeout(observe,0));
  }else{
    setTimeout(observe,0);
  }

  window.addEventListener('load',()=>{
    setTimeout(patch,150);
    setTimeout(patch,500);
    setTimeout(patch,1200);
  });
})();

  function mergeInMemory(source,target,all,visited=new Set()){
