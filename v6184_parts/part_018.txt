<script id="ux-taxonomy-level-colors-script-v6166">
(function(){
  function clean(value){return String(value??'').trim();}

  function taxonomyTerms(){
    try{
      return Array.isArray(taxonomyProject?.state?.terms) ? taxonomyProject.state.terms : [];
    }catch(_){
      return [];
    }
  }

  function termId(term){
    return clean(term?.id || term?.termId || term?.key);
  }

  function levelOf(term){
    const level=clean(term?.level).toLowerCase();
    return ['product','funnel','minifunnel','component'].includes(level) ? level : '';
  }

  function applyLevelColors(){
    const terms=taxonomyTerms();
    if(!terms.length)return;

    const byId=new Map(
      terms.map(term=>[termId(term),term]).filter(([id])=>id)
    );

    document.querySelectorAll('#taxonomy [data-taxonomy-term]').forEach(node=>{
      const term=byId.get(clean(node.dataset.taxonomyTerm));
      const level=levelOf(term);
      if(level)node.dataset.taxonomyLevel=level;
      else delete node.dataset.taxonomyLevel;
    });
  }

  let queued=false;
  function schedule(){
    if(queued)return;
    queued=true;
    requestAnimationFrame(()=>{
      queued=false;
      applyLevelColors();
    });
  }

  document.addEventListener('DOMContentLoaded',()=>{
    schedule();
    setTimeout(schedule,250);
    setTimeout(schedule,900);

    const tree=document.getElementById('taxonomy-tree');
    if(tree){
      new MutationObserver(schedule).observe(tree,{childList:true,subtree:true});
    }
  });

  document.addEventListener('click',event=>{
    if(event.target.closest('#taxonomy [data-taxonomy-term],#taxonomy [data-taxonomy-view],[data-taxonomy-manager-v2]')){
      setTimeout(schedule,0);
      setTimeout(schedule,120);
    }
  });
})();
</script>

<script id="ux-home-component-naming-migration-v6171">
(function(){
  const norm=value=>String(value??'').trim().normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLocaleLowerCase('es');
  function canonical(value){
    const key=norm(value);
    if(['hero','hero banner','hero / contenido comercial'].includes(key))return 'Hero Banner';
    if(key==='contenido comercial')return 'Claim';
    return String(value??'').trim();
  }
  function update(item){
    if(!item||typeof item!=='object'||norm(item.product)!=='home')return false;
    let changed=false;
    const next=canonical(item.component);
    if(next&&next!==String(item.component??'').trim()){item.component=next;changed=true;}
    if(Array.isArray(item.components)){
      const values=[];
      item.components.forEach(value=>{
        const mapped=canonical(value);
        if(mapped&&!values.some(existing=>norm(existing)===norm(mapped)))values.push(mapped);
      });
      if(JSON.stringify(values)!==JSON.stringify(item.components)){item.components=values;changed=true;}
    }
    return changed;
  }
  function apply(){
    let changed=false;
    const touch=item=>{if(update(item))changed=true;};
    try{(publicDemoRows||[]).forEach(touch);}catch(_){}
    try{(sourceItems||[]).forEach(doc=>{touch(doc);(doc?.insightRegister?.rows||[]).forEach(touch);});}catch(_){}
    try{(knowledgeUnits||[]).forEach(touch);}catch(_){}
    try{(learningState||[]).forEach(touch);}catch(_){}
    try{globalThis.UXSimpleKnowledge?.currentNormalizedRecords?.().forEach(touch);}catch(_){}
    if(changed){
      try{safeStorageWrite?.({project:false});}catch(_){}
    }
  }
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',()=>setTimeout(apply,0));
  else setTimeout(apply,0);
  window.addEventListener('load',()=>setTimeout(apply,500),{once:true});
})();
</script>
