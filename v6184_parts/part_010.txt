    function migrateWorkspaceToSchema3AfterRestore(){
      const persisted=safeStorageRead();
      const previousVersion=Number(persisted.workspaceSchemaVersion||0);
      if(previousVersion>=3)return false;

      let changed=false;
      const currentIds=new Set(learnings.map(item=>String(item?.id||'')));
      const hasKnownBase=[...currentIds].some(id=>id.startsWith('L-BUS-')||id.startsWith('L-HEAD-'));
      const hasSumaSource=sourceItems.some(item=>{
        const initiatives=[...(item?.initiatives||[]),item?.initiative].filter(Boolean);
        return item?.simpleKnowledgeActive&&initiatives.some(value=>normalizeText(value)===normalizeText('Usuarios SUMA'));
      });

      // This is an upgrade repair, not runtime seed behavior.
      // Only recover SUMA learnings when the structured SUMA source is actually present.
      if(hasKnownBase&&hasSumaSource){
        defaultSumaLearnings.forEach(item=>{
          if(currentIds.has(item.id))return;
          learnings.push(JSON.parse(JSON.stringify(item)));
          currentIds.add(item.id);
          changed=true;
        });
      }

      // Repair initiative metadata on the known consolidated learnings.
      learnings.forEach(item=>{
        const canonical=schema3LearningInitiatives[item.id];
        if(!canonical)return;
        const current=[...(item.initiatives||[])];
        if(JSON.stringify(current)!==JSON.stringify(canonical)){
          item.initiatives=[...canonical];
          changed=true;
        }
        if(!canonical.includes(item.initiative)){
          item.initiative=canonical[0];
          changed=true;
        }
      });

      // Persist schema 3 even if no data changed so this migration is truly one-shot.
      safeStorageWrite({project:false});
      return changed;
    }

    function rollbackLegacyGlobalQuestionReconciliation(){
      let changed=false;

      // Restaura preguntas de learnings modificadas por la reconciliación global anterior.
      learnings.forEach(learning=>{
        const history=Array.isArray(learning?.questionHistory)?learning.questionHistory:[];
        if(!history.length)return;

        const current=Array.isArray(learning.open)?[...learning.open]:[];
        const removeKeys=new Set(
          history.map(entry=>normalizeText(entry?.to||'')).filter(Boolean)
        );
        const restored=current.filter(value=>!removeKeys.has(normalizeText(value)));

        history.forEach(entry=>{
          const original=String(entry?.from||'').trim();
          if(original&&!restored.some(value=>normalizeText(value)===normalizeText(original))){
            restored.push(original);
          }
        });

        learning.open=restored;
        delete learning.questionHistory;
        changed=true;
      });

      // Las preguntas de análisis/documento son histórico: nunca deben cerrarse por matching global.
      openQuestionItems.forEach(question=>{
        const history=Array.isArray(question?.questionHistory)?question.questionHistory:[];
        if(!history.length)return;
        const original=String(history[0]?.from||'').trim();
        if(original)question.text=original;
        delete question.questionHistory;
        delete question.resolvedAt;
        delete question.resolvedBy;
        changed=true;
      });

      return changed;
    }

    function reconcileConfirmedLearningQuestions(){

    }
