    function taxonomyTermIsGoverned(term){
      if(!term)return false;
      if(['screen','element'].includes(String(term.level||'').trim()))return false;
      const status=normalizeText(term.status||'canonical');
      if(!['canonical','canonico'].includes(status))return false;
      if(term.governed===true)return true;
      const origin=String(term.origin||'').trim();
      return !/^excel revisado\s*·/i.test(origin);
    }
    function governedTaxonomyTerms(level=''){
      try{
        const terms=Array.isArray(taxonomyProject?.state?.terms)?taxonomyProject.state.terms:[];
        return terms.filter(term=>taxonomyTermIsGoverned(term)&&(!level||term.level===level));
      }catch(_){return [];}

    function governedTermName(term){return String(term?.canonicalName||term?.name||term?.label||'').trim();}
    function governedTermAliases(term){return Array.isArray(term?.aliases)?term.aliases.map(value=>String(value||'').trim()).filter(Boolean):[];}
    function governedTermMatches(term,value){
      const target=normalizeText(value||'');if(!target)return false;
      return [governedTermName(term),...governedTermAliases(term)].some(candidate=>normalizeText(candidate)===target);
    }
    function governedTermContext(term){
      if(!term)return {};
      const terms=governedTaxonomyTerms();
      const byId=new Map(terms.map(item=>[String(item.id||''),item]).filter(([id])=>id));
      const chain=[];const seen=new Set();let current=term;
      while(current){
        const id=String(current.id||'');
        if(id&&seen.has(id))break;
        if(id)seen.add(id);
        chain.unshift(current);
        const pid=String(current.parentId||'');
        current=pid?byId.get(pid):null;
      }
      const context={};
      chain.forEach(node=>{
        const level=String(node.level||'');
        const name=governedTermName(node);
        if(name&&['product','funnel','minifunnel','component'].includes(level))context[level]=name;
      });
      if(term.transversal===true){
        Object.keys(context).forEach(level=>{if(level!==term.level)delete context[level];});
      }
      return context;
    }
    function isLearningCatalogItem(item){

      return /^L(?:-|$)/.test(id)||status==='learning consolidado';

    function governedTaxonomyRecord(item){
      const source=item||{};
      // Si la taxonomía gobernada aún no ha cargado, mostramos la clasificación
      // existente. El canon normaliza cuando está disponible, pero nunca vacía
      // metadata válida por una carrera de inicialización.
      if(!governedTaxonomyTerms().length){
        const raw={
          product:String(source.product||'').trim(),
          funnel:String(source.funnel||'').trim(),
          minifunnel:String(source.minifunnel||'').trim(),
          component:String(source.component||'').trim()
        };
        if(normalizeText(raw.product)==='home'){raw.funnel='';raw.minifunnel='';}
        if(normalizeText(raw.funnel)!=='extras')raw.minifunnel='';
        return {...source,...raw};
      }
      const resolved={product:'',funnel:'',minifunnel:'',component:''};

      const candidates=(level,value)=>governedTaxonomyTerms(level).filter(term=>governedTermMatches(term,value));
      const compatible=(term,expected={})=>{
        const context=governedTermContext(term);
        return Object.entries(expected).every(([level,value])=>
          !String(value||'').trim()||normalizeText(context[level]||'')===normalizeText(value)
        );
      };
      const match=(level,value,expected={})=>{
        if(!String(value||'').trim())return null;
        const all=candidates(level,value);
        if(!all.length)return null;
        const scoped=all.filter(term=>compatible(term,expected));
        if(scoped.length)return scoped[0];
        return all.length===1?all[0]:null;
      };
      const applyContext=term=>{
        const context=governedTermContext(term);
        Object.entries(context).forEach(([key,value])=>{
          if(['product','funnel','minifunnel','component'].includes(key)&&!resolved[key])resolved[key]=value;
        });
      };

      const productTerm=match('product',source.product);
      if(productTerm)applyContext(productTerm);

      const funnelTerm=match('funnel',source.funnel,{product:resolved.product||source.product});
      if(funnelTerm)applyContext(funnelTerm);

      if(normalizeText(resolved.funnel||source.funnel)==='extras'){
        const miniTerm=match('minifunnel',source.minifunnel,{
          product:resolved.product||source.product,
          funnel:resolved.funnel||source.funnel
        });
        if(miniTerm)applyContext(miniTerm);
      }

      const rawComponents=isLearningCatalogItem(source)
        ? [source.component].filter(Boolean)
        : [...(Array.isArray(source.components)?source.components:[]),source.component].filter(Boolean);

      for(const value of rawComponents){
        const componentTerm=match('component',value,{
          product:resolved.product||source.product,
          funnel:resolved.funnel||source.funnel,
          minifunnel:resolved.minifunnel||source.minifunnel
        });
        if(componentTerm){applyContext(componentTerm);break;}
      }

      if(!resolved.product&&source.component){
        const unique=candidates('component',source.component);
        if(unique.length===1)applyContext(unique[0]);
      }

      if(normalizeText(resolved.product)==='home'){
        resolved.funnel='';
        resolved.minifunnel='';
      }
      if(normalizeText(resolved.funnel)!=='extras')resolved.minifunnel='';

      const legacy={};
      for(const level of ['product','funnel','minifunnel','component']){
        const raw=String(source[level]||'').trim();
        const canonical=String(resolved[level]||'').trim();
        if(raw&&normalizeText(raw)!==normalizeText(canonical)){
          const key='legacy'+level.charAt(0).toUpperCase()+level.slice(1);
          legacy[key]=source[key]||raw;
        }
      }
      return {...source,...legacy,...resolved};
    }
    function governedComponentNames(item){
      const learning=isLearningCatalogItem(item);

      // Un learning consolidado representa un único componente vigente.
      // Si existe `component`, NO mezclamos `legacyComponent` ni antiguos `components[]`
      // en UI, filtros o rutas. Esos valores históricos pueden seguir existiendo
      // para trazabilidad, pero no forman parte de la clasificación actual.
      if(learning){
        const current=String(item?.component||'').trim();

        // "Beneficio de fidelización" es una etiqueta conceptual histórica,
        // no un componente canónico. En el learning transversal SUMA no debe
        // mostrarse ni participar en filtros.
        if(
          String(item?.id||'')==='L-SUMA-03' ||
          (
            normalizeText(item?.funnel||'')==='transversal' &&
            normalizeText(current)==='beneficio de fidelizacion'
          )
        ){
          return [];
        }

        if(current){
          const terms=governedTaxonomyTerms('component');
          const currentContext=governedTaxonomyRecord(item);
          const compatibleTerm=terms.find(candidate=>{
            if(!governedTermMatches(candidate,current))return false;
            const context=governedTermContext(candidate);
            const same=(a,b)=>!String(b||'').trim()||normalizeText(a||'')===normalizeText(b);
            return same(context.product,currentContext.product||item?.product)&&
              same(context.funnel,currentContext.funnel||item?.funnel)&&
              same(context.minifunnel,currentContext.minifunnel||item?.minifunnel);
          });

          // Regla estricta: cuando la taxonomía gobernada está disponible,
          // un learning solo puede mostrar un componente que exista en el canon
          // y sea compatible con su contexto. Nunca inventamos ni reutilizamos
          // una etiqueta histórica como clasificación actual.
          if(terms.length){
            return compatibleTerm?[governedTermName(compatibleTerm)||current]:[];
          }

          // Antes de que el canon termine de cargar, conservamos temporalmente
          // el valor actual para evitar parpadeos; se revalida al refrescar filtros/render.
          return [current];
        }

        // Solo si falta realmente el componente actual recuperamos un histórico,
        // priorizando un valor gobernado y sin devolver más de uno.
        const fallback=[item?.legacyComponent,...(Array.isArray(item?.components)?item.components:[])]

    function taxonomyContexts(){
      const levels=['product','funnel','minifunnel','component'];
      const clean=value=>String(value??'').trim();
      const termId=term=>clean(term?.id||term?.termId||term?.key);
      const parentId=term=>clean(term?.parentId||term?.parentTermId||term?.parent?.id||term?.parent);
      const termName=term=>clean(term?.canonicalName||term?.name||term?.label||term?.value);
      const status=term=>normalizeText(term?.status||'canonical');
      const isCanonical=term=>['canonical','canonico'].includes(status(term));

      try{
        const terms=Array.isArray(taxonomyProject?.state?.terms)
          ? taxonomyProject.state.terms.filter(term=>isCanonical(term)&&taxonomyTermIsGoverned(term))
          : [];
        if(terms.length){
          const byId=new Map(terms.map(term=>[termId(term),term]).filter(([id])=>id));
          const contexts=new Map();
          terms.forEach(term=>{
            const chain=[];
            const seen=new Set();
            let current=term;
            while(current){
              const id=termId(current);
              if(id&&seen.has(id))break;
              if(id)seen.add(id);
              chain.unshift(current);
              const pid=parentId(current);
              current=pid?byId.get(pid):null;
            }
            const context={};
            chain.forEach(node=>{
              const level=clean(node?.level);
              const name=termName(node);
              if(levels.includes(level)&&name)context[level]=name;
            });
            if(!Object.keys(context).length)return;
            const key=levels.map(level=>normalizeText(context[level]||'')).join('|');
            contexts.set(key,context);
          });
          if(contexts.size)return [...contexts.values()];
        }
      }catch(_){}

      // Sin taxonomía gobernada cargada no inferimos estructura desde datos históricos.
      // El canon es fuente de verdad; los registros nunca crean términos por sí solos.
      return [];
    }

    function shouldApplyInsightRegisterFileDate(item,inferred){
      if(!inferred)return false;
      const current=temporalValue(item?.documentDate||item?.exactDate);
      if(!current)return true;
      if(current.value===inferred.value&&current.precision===inferred.precision)return false;
      if(item?.documentDate?.source!=='manual')return true;
      // A previous metadata edit could accidentally convert the ingestion month into a
      // "manual" document date even when the user only edited taxonomy/scope. If the
      // current manual date is exactly the ingestion month and the filename carries an
      // explicit document date, prefer the explicit filename date once.
      const ingestion=temporalValue(item?.ingestedAt);
      if(!ingestion)return false;
      return String(current.value||'').slice(0,7)===String(ingestion.value||'').slice(0,7);
    }
