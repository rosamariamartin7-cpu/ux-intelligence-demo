    function workspaceComparable(value,{ignoreTimestamps=true}={}){
      const ignored=ignoreTimestamps?new Set(['createdAt','updatedAt','attachedAt','ingestedAt','exportedAt']):new Set();
      const normalize=input=>{
        if(Array.isArray(input))return input.map(normalize);
        if(input&&typeof input==='object'){
          const output={};
          Object.keys(input).sort().forEach(key=>{if(!ignored.has(key))output[key]=normalize(input[key]);});
          return output;
        }
        return input??null;
      };
      return JSON.stringify(normalize(value));
    }
    function workspaceDocumentComparable(item){
      const normalized={...(item||{})};
      normalized.funnel=String(item?.funnel||'');
      normalized.minifunnel=String(item?.minifunnel||'');
      normalized.documentScopeMode=String(item?.documentScopeMode||'');
      normalized.components=Array.isArray(item?.components)?item.components:[];
      normalized.initiatives=Array.isArray(item?.initiatives)?item.initiatives:[];

    function workspaceEntityDiff(currentItems,nextItems,comparable=item=>workspaceComparable(item)){
      const currentMap=new Map((currentItems||[]).map(item=>[String(item?.id||''),item]).filter(([id])=>id));
      const nextMap=new Map((nextItems||[]).map(item=>[String(item?.id||''),item]).filter(([id])=>id));
      const added=[...nextMap.keys()].filter(id=>!currentMap.has(id));
      const removed=[...currentMap.keys()].filter(id=>!nextMap.has(id));
      const modified=[...currentMap.keys()].filter(id=>nextMap.has(id)&&comparable(currentMap.get(id))!==comparable(nextMap.get(id)));
      return {added,removed,modified,currentMap,nextMap};
    }

    function workspaceInitiativeMembership(state){
      const membership=new Map(),add=(name,member)=>{
        const value=String(name||'').trim();if(!value||normalizeText(value)==='sin clasificar'||normalizeText(value)==='documentacion')return;
        if(!membership.has(value))membership.set(value,new Set());
        membership.get(value).add(member);
      };
      (state?.documents||[]).forEach(item=>{
        const names=[item?.initiative,...(item?.initiatives||[])];
        [...new Set(names.filter(Boolean))].forEach(name=>add(name,`D:${item.id}`));
      });
      (state?.learningState||[]).forEach(item=>{
        const names=[item?.initiative,...(item?.initiatives||[])];
        [...new Set(names.filter(Boolean))].forEach(name=>add(name,`L:${item.id}`));
      });
      return membership;
    }
    function inferWorkspaceInitiativeRenames(currentState,nextState){
      const current=workspaceInitiativeMembership(currentState),next=workspaceInitiativeMembership(nextState);
      const removed=[...current.keys()].filter(name=>![...next.keys()].some(candidate=>normalizeText(candidate)===normalizeText(name)));
      const added=[...next.keys()].filter(name=>![...current.keys()].some(candidate=>normalizeText(candidate)===normalizeText(name)));
      const signature=set=>[...set].sort().join('|'),renames=[];
      removed.forEach(from=>{
        const match=added.find(to=>signature(current.get(from))&&signature(current.get(from))===signature(next.get(to))&&!renames.some(item=>item.to===to));
        if(match)renames.push({from,to:match});
      });
      return renames;
    }
    function workspaceDocumentScope(item){
      if(!item)return 'Sin clasificar';
      if(normalizeText(item.scope)==='transversal'||item.documentScopeMode==='transversal')return [item.product,'Transversal'].filter(Boolean).join(' › ');
      const raw=[item.product,item.funnel,item.minifunnel,item.component].map(value=>String(value||'').trim()).filter(Boolean),parts=[];
      raw.forEach(value=>{if(!parts.length||normalizeText(parts[parts.length-1])!==normalizeText(value))parts.push(value);});
      return parts.join(' › ')||'Sin clasificar';
    }
    function workspaceClassificationChanges(documentDiff){
      const keys=['initiative','initiatives','product','funnel','minifunnel','screen','component','element','scope','documentScopeMode'];
      return documentDiff.modified.map(id=>{
        const current=documentDiff.currentMap.get(id),next=documentDiff.nextMap.get(id);
        const changed=keys.some(key=>workspaceComparable(current?.[key])!==workspaceComparable(next?.[key]));
        if(!changed)return null;
        const currentInitiative=String(current?.initiative||(current?.initiatives||[]).join(' · ')||'Sin iniciativa');
        const nextInitiative=String(next?.initiative||(next?.initiatives||[]).join(' · ')||'Sin iniciativa');
        const initiativeChanged=normalizeText(currentInitiative)!==normalizeText(nextInitiative);
        const currentScope=workspaceDocumentScope(current),nextScope=workspaceDocumentScope(next);
        const scopeChanged=normalizeText(currentScope)!==normalizeText(nextScope);
        return {id,name:next?.name||current?.name||id,currentInitiative,nextInitiative,initiativeChanged,currentScope,nextScope,scopeChanged};
      }).filter(Boolean);
    }

    function learningSuggestionSpecificAnchorScore(row,learning){
      const anchor=String(learning?.element||'').trim();
      if(!anchor)return 1;

    function learningSuggestionScore(entry,learning,item){
      const row=entry.row,text=learningSuggestionRowText(row),learningText=learningSuggestionLearningText(learning);
      const titleScore=learningSuggestionTitleScore(row,learning);
      let score=semanticSimilarity(text,learningText)+Math.min(.18,titleScore*.28);
      if(learningSuggestionInitiativeMatches(learning,item,row))score+=.04;
      if(row?.product&&learning?.product&&normalizeText(row.product)===normalizeText(learning.product))score+=.02;
      if(row?.screen&&learning?.screen&&normalizeText(row.screen)===normalizeText(learning.screen))score+=.01;
      const rowComponent=normalizeText(row?.component||''),learningComponent=normalizeText(learning?.component||'');
      if(rowComponent&&learningComponent&&rowComponent===learningComponent)score+=.04;
      const rowContext=[row?.minifunnel,row?.component,row?.element,row?.uxLine].filter(Boolean).join(' ');
      score+=Math.min(.04,semanticSimilarity(rowContext,learningText)*.16);

      // Los learnings ligados a un elemento concreto (calendario, CTA, última búsqueda...)
      // necesitan evidencia temática real; compartir Home/Buscador no basta.
      const anchorScore=learningSuggestionSpecificAnchorScore(row,learning);
      if(learning?.element&&anchorScore<.16)score=Math.min(score,.29);

      // Lo mismo para componentes específicos: "Accesos rápidos" no debe absorber
      // registros genéricos de Buscador o Above the fold.
      const genericComponents=new Set(['','buscador','home','multipantalla','transversal']);
      if(learningComponent&&!genericComponents.has(learningComponent)){
        const componentScore=semanticSimilarity(String(row?.component||''),String(learning?.component||''));
        if(componentScore<.35&&titleScore<.35)score=Math.min(score,.29);
      }
      return Math.max(0,Math.min(1,score));
    }
    function learningSuggestionIsContradiction(row){
      return explicitContradictionSignal([learningSuggestionRowText(row),row?.observations,row?.recordType].filter(Boolean).join(' · '));
    }
    function learningSuggestionServiceTheme(entry){
      const text=normalizeText(learningSuggestionRowText(entry.row));
      const context=normalizeText([entry.row?.minifunnel,entry.row?.screen,entry.row?.component].filter(Boolean).join(' '));
      if(/asiento|equipaje|priority|fast track|sala vip/.test(context))return false;
      return /reduccion de esfuerzo|reducir esfuerzo|prerrellen|autocomplet|evita volver a introducir|evitar repetir|tareas repet|conveniencia|reconocimiento|personalizacion|capa de servicio|informacion ya conocida/.test(text);
    }
    function learningSuggestionGovernanceTheme(entry){
      const row=entry.row||{};
      const text=normalizeText([row.insight,row.uxImplication,row.uxLine,row.component].filter(Boolean).join(' '));
      const recordType=normalizeText(row.recordType||'');
      if(recordType==='restriccion')return true;
      return /fuente de verdad|gobernanza|dependencia tecnica|restriccion tecnica|amadeus|pss|contenido desactualizado|reglas.*sistema|responsable.*regla/.test(text);
    }
    function learningSuggestionCandidateId(targetKey,entries){return `LS-${learningSuggestionHash(`${targetKey}|${learningSuggestionSourceRows(entries).join(',')}`)}`;}
    function learningSuggestionExistingAction(entries,learning,scores){
      if(entries.some(entry=>learningSuggestionIsContradiction(entry.row)))return 'Contradice learning';
      const contextText=learningSuggestionLearningText(learning);
      const expanded=entries.filter(entry=>semanticSimilarity([entry.row?.component,entry.row?.screen,entry.row?.uxLine].filter(Boolean).join(' '),contextText)<.10).length;
      const average=scores.reduce((sum,value)=>sum+value,0)/Math.max(1,scores.length);
      return expanded/Math.max(1,entries.length)>.45||average<.36?'Ampliar learning':'Reforzar learning existente';
    }
    function learningSuggestionNewTitle(item,theme,entries){
      if(theme==='service-effort'){
        const initiative=String(item?.initiative||entries[0]?.row?.initiative||'').trim();
        if(/suma|frequent flyer/i.test(initiative))return 'Si ya sabemos quién es el usuario, no deberíamos pedirle lo mismo otra vez';
        return 'Si ya tenemos información del usuario, debemos usarla para evitar pasos repetidos';
      }
      const representative=String(entries[0]?.row?.insight||entries[0]?.row?.uxImplication||'Nuevo aprendizaje a consolidar').trim();
      return representative.length>120?`${representative.slice(0,117).trim()}…`:representative;
    }
    function learningSuggestionNewSummary(theme,entries){
      if(theme==='service-effort')return 'Cuando el sistema ya dispone de datos del usuario, debemos aprovecharlos para evitar repeticiones y aplicar las ventajas disponibles.';
      return entries.slice(0,3).map(entry=>String(entry.row?.uxImplication||entry.row?.insight||'').trim()).filter(Boolean).join(' ');
    }
    function learningSuggestionGenericClusters(entries){
      const clusters=[];
      entries.forEach(entry=>{
        let best=null,bestScore=0;
        clusters.forEach(cluster=>{
          const score=Math.max(...cluster.entries.map(other=>semanticSimilarity(learningSuggestionRowText(entry.row),learningSuggestionRowText(other.row))));
          if(score>bestScore){bestScore=score;best=cluster;}
        });
        if(best&&bestScore>=.24)best.entries.push(entry);else clusters.push({entries:[entry]});
      });
      return clusters;
    }
    function sameLearningSuggestionRows(left,right){
      const a=[...(left||[])].map(String).sort(),b=[...(right||[])].map(String).sort();
      return a.length===b.length&&a.every((value,index)=>value===b[index]);
    }
    function learningSuggestionResolvedRowRefs(item,state=null){
      const resolved=new Set(),register=item?.insightRegister||{},current=state||register.learningSuggestions||{};
      const addRefs=entry=>{
        if(!entry||!entry.status||entry.status==='pending')return;
        (entry.rowRefs||[]).forEach(value=>resolved.add(String(value)));
      };
      (current.history||[]).forEach(addRefs);
      (current.candidates||[]).forEach(addRefs);
      learningDecisionLogRead().forEach(entry=>{
        if(String(entry?.documentId||'')===String(item?.id||'')&&entry?.status&&entry.status!=='pending'){
          (entry.rowRefs||[]).forEach(value=>resolved.add(String(value)));
        }
      });
      try{
        learnings.forEach(learning=>{
          (learning.publicationHistory||[]).forEach(entry=>{
            if(String(entry?.documentId||'')===String(item?.id||'')){
              (entry.rowRefs||[]).forEach(value=>resolved.add(String(value)));
            }
          });
        });
      }catch(_){}
      return resolved;
    }

    function restoreLearningSuggestionCandidate(candidate,previousCandidates,item){
      const previous=(previousCandidates||[]).find(entry=>entry.fingerprint===candidate.fingerprint)
        ||(previousCandidates||[]).find(entry=>sameLearningSuggestionRows(entry.rowRefs,candidate.rowRefs)&&(entry.edited||(entry.status&&entry.status!=='pending')));
      if(previous){
        const restored={...candidate,...previous,fingerprint:candidate.fingerprint,rowRefs:candidate.rowRefs};
        if(previous.status&&previous.status!=='pending')return restored;
        if(previous.edited)return restored;
      }
      const rowsKey=learningDecisionRowsKey(candidate.rowRefs);
      const remembered=[...learningDecisionLogRead()].reverse().find(entry=>
        String(entry.documentId||'')===String(item?.id||'')&&
        ((rowsKey&&entry.rowsKey===rowsKey)||(candidate.fingerprint&&entry.fingerprint===candidate.fingerprint))
      );
      if(remembered&&remembered.status&&remembered.status!=='pending'){
        return {
          ...candidate,
          status:remembered.status,
          action:remembered.action||candidate.action,
          targetLearningId:remembered.learningId||candidate.targetLearningId,
          appliedLearningId:remembered.learningId||candidate.appliedLearningId,
          title:remembered.title||candidate.title,
          know:remembered.know||candidate.know,
          text:remembered.text||candidate.text,
          edited:Boolean(remembered.edited||candidate.edited),
          resolvedAt:remembered.savedAt||candidate.resolvedAt||''
        };
      }
      try{
        for(const learning of learnings){
          const publication=(learning.publicationHistory||[]).find(entry=>
            String(entry?.documentId||'')===String(item?.id||'')&&
            sameLearningSuggestionRows(entry?.rowRefs,candidate.rowRefs)
          );
          if(publication){
            return {
              ...candidate,
              status:'confirmed',
              action:publication.decision||candidate.action,
              targetLearningId:learning.id,
              appliedLearningId:learning.id,
              resolvedAt:publication.publishedAt||candidate.resolvedAt||''
            };
          }
        }
      }catch(_){}
      return candidate;
    }
    function buildLearningSuggestions(item,previousState=null){
      const register=item?.insightRegister;if(!register?.rows?.length)return null;
      const allEntries=register.rows.map((row,index)=>({row,index}));
      const rowFingerprint=learningSuggestionHash(allEntries.map(entry=>learningSuggestionRowSignature(entry.row,entry.index)).join('\n'));
      const previousCandidates=previousState?.candidates||register.learningSuggestions?.candidates||[];
      const resolvedRows=learningSuggestionResolvedRowRefs(item,previousState||register.learningSuggestions);
      const entries=allEntries.filter(entry=>!resolvedRows.has(String(entry.row?.sourceRow||entry.index+1)));
      const resolvedCandidates=(previousCandidates||[]).filter(candidate=>candidate?.status&&candidate.status!=='pending');
      const historicalResolved=[];
      (previousState?.history||register.learningSuggestions?.history||[]).forEach(historyEntry=>{
        if(!historyEntry?.status||historyEntry.status==='pending')return;
        const alreadyPresent=resolvedCandidates.some(candidate=>
          sameLearningSuggestionRows(candidate?.rowRefs,historyEntry?.rowRefs)&&
          String(candidate?.status||'')===String(historyEntry.status||'')&&
          String(candidate?.targetLearningId||candidate?.appliedLearningId||'')===String(historyEntry?.learningId||'')
        );
        if(alreadyPresent)return;
        const refs=new Set((historyEntry.rowRefs||[]).map(String));
        const historicalEntries=allEntries.filter(entry=>refs.has(String(entry.row?.sourceRow||entry.index+1)));
        const action=historyEntry.decision||(historyEntry.status==='evidence-only'?'Mantener solo como evidencia':'Reforzar learning existente');
        historicalResolved.push({
          action,
          targetLearningId:historyEntry.learningId||'',
          appliedLearningId:historyEntry.learningId||'',
          newLearningTitle:'',
          summary:historyEntry.correctionReason||'Decisión ya resuelta y conservada en el historial.',
          impactUx:historicalEntries.length?learningSuggestionImpact(historicalEntries):'Medio',
          rationale:historyEntry.correctionReason||'Decisión histórica ya revisada.',
          evidence:historicalEntries.length?learningSuggestionEvidence(historicalEntries,item):[],
          theme:historyEntry.status==='evidence-only'?'evidence-only':'',
          id:historyEntry.candidateId||learningSuggestionCandidateId(historyEntry.learningId||action,historicalEntries),
          fingerprint:historyEntry.fingerprint||learningSuggestionHash(`${historyEntry.learningId||action}|${(historyEntry.rowRefs||[]).join(',')}`),
          rowRefs:[...(historyEntry.rowRefs||[])],
          status:historyEntry.status,
          resolvedAt:historyEntry.at||'',
          createdAt:historyEntry.at||new Date().toISOString()
        });
      });
      const reserved=new Set(),candidates=[...resolvedCandidates,...historicalResolved];
      const addCandidate=(candidate,entriesForCandidate)=>{
        const rowRefs=learningSuggestionSourceRows(entriesForCandidate),targetKey=candidate.targetLearningId||candidate.theme||candidate.action;
        const fingerprint=learningSuggestionHash(`${targetKey}|${rowRefs.join(',')}`);
        candidates.push(restoreLearningSuggestionCandidate({...candidate,id:learningSuggestionCandidateId(targetKey,entriesForCandidate),fingerprint,rowRefs,status:'pending',createdAt:new Date().toISOString()},previousCandidates,item));
        entriesForCandidate.forEach(entry=>reserved.add(entry.index));
      };

      const serviceEntries=entries.filter(learningSuggestionServiceTheme);
      if(serviceEntries.length>=3)addCandidate({action:'Crear nuevo learning',theme:'service-effort',targetLearningId:'',newLearningTitle:learningSuggestionNewTitle(item,'service-effort',serviceEntries),summary:learningSuggestionNewSummary('service-effort',serviceEntries),impactUx:learningSuggestionImpact(serviceEntries),rationale:`${serviceEntries.length} registros repiten la misma idea: usar información ya conocida para reducir pasos y esfuerzo.`,evidence:learningSuggestionEvidence(serviceEntries,item)},serviceEntries);

      const governanceEntries=entries.filter(entry=>!reserved.has(entry.index)&&learningSuggestionGovernanceTheme(entry));
      const governanceIndexes=new Set(governanceEntries.map(entry=>entry.index));

      const groups=new Map(),unmatched=[];
      entries.filter(entry=>!reserved.has(entry.index)&&!governanceIndexes.has(entry.index)).forEach(entry=>{
        const ranked=learnings.map(learning=>({learning,score:learningSuggestionScore(entry,learning,item)})).sort((a,b)=>b.score-a.score);
        const best=ranked[0];
        const titleScore=best?learningSuggestionTitleScore(entry.row,best.learning):0;
        const anchorScore=best?learningSuggestionSpecificAnchorScore(entry.row,best.learning):0;
        const qualifies=best&&(best.score>=.34||titleScore>=.22||(best.learning?.element&&anchorScore>=.35));
        if(qualifies){
          if(!groups.has(best.learning.id))groups.set(best.learning.id,{learning:best.learning,entries:[],scores:[]});
          groups.get(best.learning.id).entries.push(entry);groups.get(best.learning.id).scores.push(best.score);
        }else unmatched.push(entry);
      });
      groups.forEach(group=>{
        const action=learningSuggestionExistingAction(group.entries,group.learning,group.scores);
        addCandidate({action,targetLearningId:group.learning.id,newLearningTitle:'',summary:`${group.entries.length} ${group.entries.length===1?'registro aporta':'registros aportan'} evidencia nueva al learning existente y ${action==='Reforzar learning existente'?'refuerza su dirección actual':'amplía su contexto o aplicaciones'}.`,impactUx:learningSuggestionImpact(group.entries),rationale:`Coincidencia con ${group.learning.id} · ${Math.round((group.scores.reduce((sum,value)=>sum+value,0)/group.scores.length)*100)}% de afinidad media.`,evidence:learningSuggestionEvidence(group.entries,item)},group.entries);
      });

      const genericClusters=learningSuggestionGenericClusters(unmatched).filter(cluster=>cluster.entries.length>=3);
      genericClusters.slice(0,2).forEach((cluster,index)=>addCandidate({action:'Crear nuevo learning',theme:`cluster-${index+1}`,targetLearningId:'',newLearningTitle:learningSuggestionNewTitle(item,`cluster-${index+1}`,cluster.entries),summary:learningSuggestionNewSummary(`cluster-${index+1}`,cluster.entries),impactUx:learningSuggestionImpact(cluster.entries),rationale:`${cluster.entries.length} registros no encajan con suficiente claridad en un learning existente y comparten una misma línea temática.`,evidence:learningSuggestionEvidence(cluster.entries,item)},cluster.entries));

      const evidenceOnly=entries.filter(entry=>!reserved.has(entry.index));
      if(evidenceOnly.length)addCandidate({action:'Mantener solo como evidencia',theme:'evidence-only',targetLearningId:'',newLearningTitle:'Registros que todavía no justifican un learning independiente',summary:'Estos registros aportan contexto, reglas u oportunidades útiles, pero todavía no hay suficiente evidencia común para convertirlos en un learning propio.',impactUx:learningSuggestionImpact(evidenceOnly),rationale:`${evidenceOnly.length} registros quedan como evidencia estructurada sin forzar una conclusión consolidada.`,evidence:learningSuggestionEvidence(evidenceOnly,item)},evidenceOnly);

      const resolvedOutput=candidates.filter(candidate=>candidate?.status&&candidate.status!=='pending');
      const pendingOutput=candidates.filter(candidate=>!candidate?.status||candidate.status==='pending').slice(0,Math.max(0,6-resolvedOutput.length));
      const state={version:LEARNING_SUGGESTION_VERSION,generatedAt:new Date().toISOString(),rowsFingerprint:rowFingerprint,candidates:[...resolvedOutput,...pendingOutput],history:[...(previousState?.history||register.learningSuggestions?.history||[])]};
      register.learningSuggestions=state;return state;
    }
    function ensureLearningSuggestions(item,{force=false,previousState=null}={}){

      return buildLearningSuggestions(item,previousState||current);
    }

    function learningHistoryRowRefs(refs){
      return [...new Set([...(refs||[])].map(value=>String(value)))];
    }
    function learningHistoryRowsNested(left,right){
      const a=learningHistoryRowRefs(left),b=learningHistoryRowRefs(right);
      if(!a.length||!b.length)return false;
      const aInB=a.every(value=>b.includes(value));
      const bInA=b.every(value=>a.includes(value));
      return aInB||bInA;
    }
    function learningHistoryWithinWindow(left,right){
      const a=Date.parse(left||''),b=Date.parse(right||'');
      return Number.isFinite(a)&&Number.isFinite(b)&&Math.abs(a-b)<=10*60*1000;
    }
    function dedupeLearningPublicationHistory(entries){
      const output=[];
      [...(entries||[])].forEach(entry=>{
        const refs=learningHistoryRowRefs(entry?.rowRefs);
        if(!refs.length||!entry?.documentId){output.push(entry);return;}
        let duplicateIndex=-1;
        for(let index=output.length-1;index>=0;index--){
          const previous=output[index];
          if(String(previous?.documentId||'')!==String(entry.documentId||''))continue;
          if(String(previous?.decision||'')!==String(entry?.decision||''))continue;
          const exactRows=sameLearningSuggestionRows(previous?.rowRefs,entry?.rowRefs);
          const nestedRecent=learningHistoryRowsNested(previous?.rowRefs,entry.rowRefs)&&learningHistoryWithinWindow(previous?.publishedAt,entry?.publishedAt);
          if(!exactRows&&!nestedRecent)continue;
          duplicateIndex=index;
          break;
        }
        if(duplicateIndex<0){output.push(entry);return;}
        const previous=output[duplicateIndex];
        const previousRefs=learningHistoryRowRefs(previous?.rowRefs);
        const previousAt=Date.parse(previous?.publishedAt||'');
        const currentAt=Date.parse(entry?.publishedAt||'');
        const preferCurrent=refs.length>previousRefs.length||(refs.length===previousRefs.length&&currentAt>=previousAt);
        if(preferCurrent){
          output.splice(duplicateIndex,1);
          output.push(entry);
        }
      });
      return output;
    }
    function dedupeLearningSuggestionHistory(entries){
      const output=[];
      [...(entries||[])].forEach(entry=>{
        const refs=learningHistoryRowRefs(entry?.rowRefs);
        if(!refs.length){output.push(entry);return;}
        let duplicateIndex=-1;
        for(let index=output.length-1;index>=0;index--){
          const previous=output[index];
          if(String(previous?.learningId||'')!==String(entry?.learningId||''))continue;
          if(String(previous?.decision||'')!==String(entry?.decision||''))continue;
          const exactRows=sameLearningSuggestionRows(previous?.rowRefs,entry?.rowRefs);
          const nestedRecent=learningHistoryRowsNested(previous?.rowRefs,entry.rowRefs)&&learningHistoryWithinWindow(previous?.at,entry?.at);
          if(!exactRows&&!nestedRecent)continue;
          duplicateIndex=index;
          break;
        }
        if(duplicateIndex<0){output.push(entry);return;}
        const previous=output[duplicateIndex];
        const previousRefs=learningHistoryRowRefs(previous?.rowRefs);
        const previousAt=Date.parse(previous?.at||'');
        const currentAt=Date.parse(entry?.at||'');
        const preferCurrent=refs.length>previousRefs.length||(refs.length===previousRefs.length&&currentAt>=previousAt);
        if(preferCurrent){
          output.splice(duplicateIndex,1);
          output.push(entry);
        }
      });
      return output;
    }
    function normalizeLearningHistoryState(){
      let changed=false;
      learnings.forEach(learning=>{
        const current=[...(learning?.publicationHistory||[])];
        const cleaned=dedupeLearningPublicationHistory(current);
        if(JSON.stringify(current)!==JSON.stringify(cleaned)){
          learning.publicationHistory=cleaned;
          changed=true;
        }
      });
      sourceItems.forEach(item=>{
        const state=item?.insightRegister?.learningSuggestions;
        if(!state)return;
        const current=[...(state.history||[])];
        const cleaned=dedupeLearningSuggestionHistory(current);
        if(JSON.stringify(current)!==JSON.stringify(cleaned)){
          state.history=cleaned;
          changed=true;
        }
      });
      return changed;
    }

    }
    function questionResolutionRowText(row){
      return [row?.insight,row?.evidence,row?.implication,row?.recommendation,row?.openQuestion,row?.product,row?.funnel,row?.minifunnel,row?.screen,row?.component,row?.element]
        .filter(Boolean).join(' · ');
    }
    function questionOwnerCompatible(owner,row){
      const ownerProduct=normalizeText(owner?.product||''),rowProduct=normalizeText(row?.product||'');
      return !(ownerProduct&&rowProduct&&ownerProduct!==rowProduct);
    }
    function questionLooksLikeDesignQuestion(question){
      const value=normalizeText(question||'');
      const asksHow=/^[¿?¡!\s]*(como|de que forma|que patron|que tratamiento|como deberia|como deben)/.test(value);
      const designTopic=/(comunic|mostrar|presentar|explicar|diseñ|patron|interaccion|mensaje|jerarquia|visual)/.test(value);
      return asksHow&&designTopic;
    }
    function questionTopicBonus(question,row){
      const q=normalizeText(question||''),text=normalizeText(questionResolutionRowText(row));
      let bonus=0;
      ['upsell','fare families','fare family','priority boarding','asientos','equipaje','login','multipasajero'].forEach(term=>{
        if(q.includes(term)&&text.includes(term))bonus+=0.10;
      });
      ['millas','acumulacion','beneficios','tarifa','suma','pasajero','elegibilidad','descuento'].forEach(term=>{
        if(q.includes(term)&&text.includes(term))bonus+=0.025;
      });
      return Math.min(0.18,bonus);
    }
    function questionResolutionMatch(question,owner,rows){
      // Las preguntas de diseño/representación siguen abiertas aunque una dependencia técnica se aclare.
      if(questionLooksLikeDesignQuestion(question))return null;
      let best=null;
      (rows||[]).forEach(row=>{
        const status=questionResolutionStatus(row);
        if(!status||!questionOwnerCompatible(owner,row))return;
        let score=semanticSimilarity(question,questionResolutionRowText(row))+questionTopicBonus(question,row);
        if(normalizeText(owner?.product||'')&&normalizeText(owner?.product||'')===normalizeText(row?.product||''))score+=0.04;
        const ownerComponent=normalizeText(owner?.component||''),rowComponent=normalizeText(row?.component||'');
        if(ownerComponent&&rowComponent&&ownerComponent===rowComponent)score+=0.04;
        const ownerScreen=normalizeText(owner?.screen||''),rowScreen=normalizeText(row?.screen||'');
        if(ownerScreen&&rowScreen&&(ownerScreen===rowScreen||rowScreen.includes(ownerScreen)||ownerScreen.includes(rowScreen)))score+=0.03;
        if(score>=0.30&&(!best||score>best.score))best={row,status,score};
      });
      return best;
    }
    function recordQuestionTransition(owner,from,to,status,item,row){
      owner.questionHistory=Array.isArray(owner.questionHistory)?owner.questionHistory:[];
      const entry={from:String(from||''),to:String(to||''),status,sourceName:item?.name||'',sourceFile:item?.originalName||'',at:new Date().toISOString(),rowRef:row?.sourceRow||''};
      const duplicate=owner.questionHistory.some(value=>normalizeText(value?.from||'')===normalizeText(entry.from)&&normalizeText(value?.to||'')===normalizeText(entry.to)&&String(value?.sourceName||'')===entry.sourceName);
      if(!duplicate)owner.questionHistory.push(entry);
    }
    function reconcileLearningQuestions(learning,rows,item){
      const current=Array.isArray(learning?.open)?learning.open:[];
      if(!current.length)return false;
      let changed=false;
      const next=[];
      current.forEach(question=>{
        const match=questionResolutionMatch(question,learning,rows);
        if(!match){next.push(question);return;}
        const replacement=String(match.row?.openQuestion||'').trim();
        if(match.status==='resolved'){
          changed=true;
          if(replacement&&normalizeText(replacement)!==normalizeText(question))next.push(replacement);
          recordQuestionTransition(learning,question,replacement,'resolved',item,match.row);
          return;
        }
        if(replacement&&normalizeText(replacement)!==normalizeText(question)){
          next.push(replacement);changed=true;
          recordQuestionTransition(learning,question,replacement,match.status,item,match.row);
        }else next.push(question);
      });
      const seen=new Set();
      learning.open=next.filter(value=>{const key=normalizeText(value);if(!key||seen.has(key))return false;seen.add(key);return true;});
      if(changed)learning.updatedAt=new Date().toISOString().slice(0,10);
      return changed;
    }
    function reconcileStandaloneQuestions(rows,item){
      let changed=false;
      openQuestionItems.forEach(questionItem=>{
        if(questionItem?.resolvedAt||!questionItem?.text)return;
        const match=questionResolutionMatch(questionItem.text,questionItem,rows);
        if(!match)return;
        const replacement=String(match.row?.openQuestion||'').trim();
        recordQuestionTransition(questionItem,questionItem.text,replacement,match.status,item,match.row);
        if(match.status==='resolved'&&!replacement){
          questionItem.resolvedAt=new Date().toISOString();
          questionItem.resolvedBy=item?.name||'';
          changed=true;
          return;
        }
        if(replacement&&normalizeText(replacement)!==normalizeText(questionItem.text)){
          questionItem.text=replacement;
          questionItem.updatedAt=new Date().toISOString();
          changed=true;
        }
      });
      return changed;
    }
    function reconcileOpenQuestionsFromRows(rows,item){
      const useful=(rows||[]).filter(row=>questionResolutionStatus(row));
      if(!useful.length)return false;
      let changed=false;
      learnings.forEach(learning=>{if(reconcileLearningQuestions(learning,useful,item))changed=true;});
      if(reconcileStandaloneQuestions(useful,item))changed=true;
      return changed;
    }
