  function firstNonEmptyValue(...values){
    for(const value of values){
      if(Array.isArray(value)){
        if(value.length)return value;
        continue;
      }
      if(clean(value))return value;
    }
    return '';
  }

  function fullExportValue(record,key){
    const reviewed=reviewedFullMeta(record)||{};

    // La clasificación estructural actual debe salir únicamente del registro gobernado.
    // Los valores históricos se conservan en sus columnas legacy, pero no se reinyectan
    // en Producto/Funnel/Minifunnel/Componente porque pueden pertenecer a taxonomías antiguas.
    if(key==='product')return record.product||'';
    if(key==='funnel')return record.funnel||'';
    if(key==='minifunnel')return record.minifunnel||'';
    if(key==='component'){
      const governed=typeof componentsOf==='function'?componentsOf(record):[];
      return firstNonEmptyValue(governed,record.component);
    }
    if(key==='recordType'){
      const current=clean(record.recordType);
      if(current)return current;
      return reviewed.sourceRecordType?normalizeRecordType(reviewed.sourceRecordType):'';
    }
    if(key==='origin'){
      const current=clean(record.origin);
      if(current)return current;
      return reviewed.sourceAnalysisType?normalizeOrigins(reviewed.sourceAnalysisType).join(' · '):'';
    }
    if(key==='uxLine'){
      // Dimensión es la clasificación UX actual. Si está vacía, se mantiene vacía.
      // La clasificación histórica sigue disponible en 'Dimensión UX legacy'.
      return clean(record.uxLine);
    }

    const reviewedFallbackKeys=new Set([
      'initiative','evidence','implication','recommendation','openQuestion',
      'platformSystem','device','market','impactUx','notes',
      'solutionId','solutionUx','designRationale','designRef','solutionStatus','solutionRelation'
    ]);
    if(reviewedFallbackKeys.has(key))return firstNonEmptyValue(record[key],reviewed[key]);

    if(key==='sourceRecordType')return reviewed.sourceRecordType||'';
    if(key==='sourceAnalysisType')return reviewed.sourceAnalysisType||'';
    if(key==='sourceRegion')return reviewed.sourceRegion||'';
    if(key==='legacyDimension')return reviewed.legacyDimension||'';
    if(key==='dimensionPrimary')return reviewed.dimensionPrimary||'';
    if(key==='dimensionSecondary')return reviewed.dimensionSecondary||'';

    return record[key];
  }

<script id="ux-taxonomy-source-of-truth-script">
(function(){
  const LEVELS=['product','funnel','minifunnel','component'];
  const LEVEL_LABELS={product:'Producto',funnel:'Funnel',minifunnel:'Minifunnel',component:'Componente'};
  let activeView='product';
  let activeTermId='';

  const clean=value=>String(value??'').trim();
  const norm=value=>clean(value).normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLocaleLowerCase('es');
  const esc=value=>String(value??'').replace(/[&<>"']/g,char=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[char]));

  function terms(){
    try{
      const all=Array.isArray(taxonomyProject?.state?.terms)?taxonomyProject.state.terms:[];
      return all.filter(term=>{
        const origin=String(term?.origin||'').trim();
        return term?.governed===true||!/^excel revisado\s*·/i.test(origin);
      });
    }catch(_){return [];}
  }

  function termId(term){return clean(term?.id||term?.termId||term?.key);}
  function parentId(term){return clean(term?.parentId||term?.parentTermId||term?.parent?.id||term?.parent);}
  function termName(term){return clean(term?.canonicalName||term?.name||term?.label||term?.value);}
  function termStatus(term){return clean(term?.status||'canonical')||'canonical';}
  function termStatusKind(term){
    const value=norm(termStatus(term));
    if(['canonical','canonico'].includes(value))return 'canonical';
    if(['alias','synonym','sinonimo'].includes(value))return 'alias';
    if(['obsolete','obsoleto','deprecated','retired','retirado'].includes(value))return 'obsolete';
    if(['pending','pendiente','proposed','proposal','propuesta','draft','borrador'].includes(value))return 'pending';
    return 'pending';
  }
  function termStatusLabel(term){
    return {canonical:'Canónico',alias:'Alias',obsolete:'Obsoleto',pending:'Pendiente'}[termStatusKind(term)]||termStatus(term);
  }
  function termLevel(term){return clean(term?.level);}
  function termAliases(term){
    const raw=term?.aliases||term?.alias||term?.historicalNames||[];
    return Array.isArray(raw)?raw.map(clean).filter(Boolean):clean(raw)?[clean(raw)]:[];
  }

  function findTerm(id){return terms().find(term=>termId(term)===clean(id));}

  function ancestry(term){
    if(!term)return [];
    if(Array.isArray(term.path)&&term.path.length)return term.path.map(clean).filter(Boolean);
    if(typeof term.path==='string'&&term.path.trim())return term.path.split(/\s*[›/>]\s*/).map(clean).filter(Boolean);
    const byId=new Map(terms().map(item=>[termId(item),item]).filter(([id])=>id));
    const chain=[];
    const seen=new Set();
    let current=term;
    while(current&&!seen.has(termId(current))){
      const id=termId(current);if(id)seen.add(id);
      chain.unshift(current);
      const pid=parentId(current);
      current=pid?byId.get(pid):null;
    }
    return chain.map(termName).filter(Boolean);
  }

  function contextForTerm(term){
    const context={};
    if(!term)return context;
    const byId=new Map(terms().map(item=>[termId(item),item]).filter(([id])=>id));
    const chain=[];
    const seen=new Set();
    let current=term;
    while(current&&!seen.has(termId(current))){
      const id=termId(current);if(id)seen.add(id);
      chain.unshift(current);
      const pid=parentId(current);
      current=pid?byId.get(pid):null;
    }
    chain.forEach(item=>{
      const level=termLevel(item);
      const name=termName(item);
      if(LEVELS.includes(level)&&name)context[level]=name;
    });
    if(!Object.keys(context).length&&LEVELS.includes(termLevel(term)))context[termLevel(term)]=termName(term);
    return context;
  }

  function structuredRecords(){
    try{
      if(typeof currentStructuredRecords==='function')return currentStructuredRecords();
    }catch(_){}
    try{
      return (sourceItems||[]).flatMap(item=>(item.insightRegister?.rows||[]).map(row=>({...row,__documentId:item.id})));
    }catch(_){return [];}
  }

  function matchesContext(item,context){
    return LEVELS.every(level=>{
      if(!context[level])return true;
      if(level==='component'){
        const values=[item?.component,...(Array.isArray(item?.components)?item.components:[])].map(norm).filter(Boolean);
        return values.includes(norm(context[level]));
      }
      return norm(item?.[level])===norm(context[level]);
    });
  }

  function knowledgeMetrics(term){
    const context=contextForTerm(term);
    const rows=structuredRecords().filter(row=>matchesContext(row,context));
    let learningItems=[];
    try{
      learningItems=(learnings||[]).filter(item=>{
        const candidates=[item,{...item,component:item.component||item.components?.[0]}];
        return candidates.some(candidate=>matchesContext(candidate,context));
      });
    }catch(_){}
    let docs=[];
    try{
      docs=(sourceItems||[]).filter(doc=>{
        const docRows=doc.insightRegister?.rows||[];
        return docRows.some(row=>matchesContext(row,context))||matchesContext(doc,context);
      });
    }catch(_){}
    return {context,records:rows.length,learnings:learningItems.length,sources:docs.length};
  }

  function simplifyTaxonomyDetail(){
    const detail=document.getElementById('taxonomy-detail');
    if(!detail)return;

    const normalized=value=>norm(String(value||'').replace(/\s+/g,' ').trim());
    const all=[...detail.querySelectorAll('*')];

    // El estado normal no necesita una etiqueta visible.
    all.filter(node=>normalized(node.textContent)==='canonico').forEach(node=>{
      const candidate=node.closest('.status-badge,.badge,.taxonomy-badge,.taxonomy-status')||node;
      candidate.hidden=true;
      candidate.setAttribute('aria-hidden','true');
    });

    // "Nivel" duplica el kicker superior (PRODUCTO / PANTALLA / COMPONENTE / ELEMENTO).
    // Dejamos visible la relación jerárquica "Depende de", que sí aporta contexto.
    const levelLabelNode=[...detail.querySelectorAll('span,small,dt,label,strong')].find(node=>normalized(node.textContent)==='nivel');
    if(levelLabelNode){
      let block=levelLabelNode.parentElement;
      while(block&&block!==detail){
        const copy=normalized(block.textContent);
        if(copy.startsWith('nivel')&&copy.length<100){
          block.hidden=true;
          break;
        }
        block=block.parentElement;
      }
    }

    // Los alias vacíos son ruido; si existen, se expresan con lenguaje más claro.
    const aliasHeading=all.find(node=>['h2','h3','h4','strong'].includes(node.tagName.toLowerCase())&&normalized(node.textContent)==='alias');
    if(aliasHeading){
      let block=aliasHeading.parentElement;
      while(block&&block!==detail){
        const copy=normalized(block.textContent);
        if(copy.includes('no tiene alias')){
          block.hidden=true;
          break;
        }

        block=block.parentElement;

    const useHeading=all.find(node=>['h2','h3','h4','strong'].includes(node.tagName.toLowerCase())&&normalized(node.textContent)==='uso actual');

            });

          break;

        block=block.parentElement;
      }
    }

            break;

      }

<script id="uikit-date-picker-script">
(function(){
  const MONTHS=['enero','febrero','marzo','abril','mayo','junio','julio','agosto','septiembre','octubre','noviembre','diciembre'];
  const WEEKDAYS=['L','M','X','J','V','S','D'];
  const pad=n=>String(n).padStart(2,'0');
  const toIso=date=>`${date.getFullYear()}-${pad(date.getMonth()+1)}-${pad(date.getDate())}`;
  const parseIso=value=>{
    if(!/^\d{4}-\d{2}-\d{2}$/.test(value||''))return null;
    const [y,m,d]=value.split('-').map(Number);
    const date=new Date(y,m-1,d,12);
    return Number.isNaN(date.getTime())?null:date;
  };
  const sameDay=(a,b)=>!!a&&!!b&&a.getFullYear()===b.getFullYear()&&a.getMonth()===b.getMonth()&&a.getDate()===b.getDate();
  const formatDate=date=>date?`${date.getDate()} ${MONTHS[date.getMonth()].slice(0,4)} ${date.getFullYear()}`:'Seleccionar fecha';
  const esc=s=>String(s??'').replace(/[&<>"']/g,ch=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[ch]));

  let activePicker=null;

  function rangePair(input){
    const id=input?.id||'';
    const base=id.replace(/-(from|to)$/,'');
    const role=id.endsWith('-to')?'to':'from';
    return {
      role,
      from:document.getElementById(`${base}-from`),
      to:document.getElementById(`${base}-to`)
    };
  }
  function beforeDay(a,b){
    return !!a&&!!b&&toIso(a)<toIso(b);
  }

  function calendarIcon(){
    return `<svg viewBox="0 0 24 24" fill="none" aria-hidden="true"><path d="M8 2v3M16 2v3M3.5 9h17M5.5 4.5h13a2 2 0 0 1 2 2v12a2 2 0 0 1-2 2h-13a2 2 0 0 1-2-2v-12a2 2 0 0 1 2-2Z" stroke="currentColor" stroke-width="1.7" stroke-linecap="round" stroke-linejoin="round"/></svg>`;
  }
  function chevronIcon(direction){
    const d=direction==='left'?'m15 18-6-6 6-6':'m9 18 6-6-6-6';
    return `<svg viewBox="0 0 24 24" fill="none" aria-hidden="true"><path d="${d}" stroke="currentColor" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round"/></svg>`;
  }

  function positionMenu(picker){
    const {trigger,menu}=picker;
    if(!menu||menu.hidden)return;
    const rect=trigger.getBoundingClientRect();
    const menuWidth=Math.min(328,window.innerWidth-32);
    menu.style.width=`${menuWidth}px`;
    menu.style.left=`${Math.min(Math.max(16,rect.left),Math.max(16,window.innerWidth-menuWidth-16))}px`;

    menu.style.top='0px';
    const menuHeight=Math.min(menu.scrollHeight,window.innerHeight-32);
    const below=window.innerHeight-rect.bottom-8;
    const top=below>=menuHeight
      ? rect.bottom+8
      : Math.max(16,rect.top-menuHeight-8);
    menu.style.top=`${top}px`;
  }

  function renderCalendar(picker){
    const {menu}=picker;
    const year=picker.viewYear,month=picker.viewMonth;
    const first=new Date(year,month,1,12);
    const mondayIndex=(first.getDay()+6)%7;
    const gridStart=new Date(year,month,1-mondayIndex,12);
    const today=new Date(); today.setHours(12,0,0,0);
    const pair=rangePair(picker.input);
    const start=parseIso(pair.from?.value||'');
    const end=parseIso(pair.to?.value||'');

    const weekdays=WEEKDAYS.map(day=>`<span class="ui-date-weekday" role="columnheader">${day}</span>`).join('');
    let days='';
    for(let i=0;i<42;i++){
      const date=new Date(gridStart);date.setDate(gridStart.getDate()+i);
      const outside=date.getMonth()!==month;
      const isStart=sameDay(date,start);
      const isEnd=sameDay(date,end);
      const selected=isStart||isEnd;
      const inRange=!!start&&!!end&&beforeDay(start,date)&&beforeDay(date,end);
      const isToday=sameDay(date,today);
      const disabled=pair.role==='to'&&!!start&&beforeDay(date,start);
      days+=`<button class="ui-date-day${outside?' is-outside':''}${inRange?' is-in-range':''}${isStart?' is-range-start':''}${isEnd?' is-range-end':''}${selected?' is-selected':''}${isToday?' is-today':''}${disabled?' is-disabled':''}" type="button" role="gridcell" aria-selected="${selected?'true':'false'}" aria-disabled="${disabled?'true':'false'}" ${disabled?'disabled':''} data-date="${toIso(date)}" aria-label="${esc(formatDate(date))}">${date.getDate()}</button>`;
    }

    menu.querySelector('.ui-date-month-label').textContent=`${MONTHS[month]} ${year}`;
    menu.querySelector('.ui-date-grid').innerHTML=weekdays+days;
  }

  function closePicker(picker,{restoreFocus=true}={}){
    if(!picker)return;
    picker.menu.hidden=true;
    picker.wrapper.classList.remove('is-open');
    picker.trigger.setAttribute('aria-expanded','false');
    if(activePicker===picker)activePicker=null;
    if(restoreFocus)picker.trigger.focus({preventScroll:true});
  }

  function openPicker(picker){
    if(activePicker&&activePicker!==picker)closePicker(activePicker,{restoreFocus:false});
    const pair=rangePair(picker.input);
    const existing=parseIso(picker.input.value);
    const start=parseIso(pair.from?.value||'');
    const end=parseIso(pair.to?.value||'');
    picker.pending=existing?new Date(existing):null;
    const base=existing||(pair.role==='to'?start:null)||start||end||new Date();
    picker.viewYear=base.getFullYear();
    picker.viewMonth=base.getMonth();
    picker.menu.hidden=false;
    picker.wrapper.classList.add('is-open');
    picker.trigger.setAttribute('aria-expanded','true');
    activePicker=picker;
    renderCalendar(picker);
    requestAnimationFrame(()=>positionMenu(picker));
  }

  function syncTrigger(picker){
    const selected=parseIso(picker.input.value);
    picker.value.textContent=formatDate(selected);
    picker.trigger.classList.toggle('is-placeholder',!selected);
  }
